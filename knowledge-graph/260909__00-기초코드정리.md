# 추출 품질 평가 기초 코드

LLM으로 추출한 트리플을 **스키마 → 근거 원문 → 사람 판정 → 골드셋** 순서로 점검하는 기본 흐름.

> 핵심 흐름: `스키마 준수 검사 → 근거 원문 일치 검사 → 판정 정밀도 → 미니 골드 exact 평가 → 결과 저장`

---

## 0. 공통 준비

### 파일 읽기·저장 함수

```python
import json
import random
from collections import Counter
from pathlib import Path

# 제공된 원문, 추출 트리플, 골드 파일 폴더
data_dir = Path("data")

# 계산한 지표와 검토 기록 저장 폴더
output_dir = Path("output")
output_dir.mkdir(exist_ok=True)


def load_rows(filename):
    """data 폴더의 JSONL 파일을 딕셔너리 목록으로 읽는다."""
    return [
        json.loads(line)
        for line in (data_dir / filename).read_text(encoding="utf-8").splitlines()
        if line.strip()
    ]


def write_json(filename, value):
    """결과를 output 폴더에 JSON으로 저장한다."""
    (output_dir / filename).write_text(
        json.dumps(value, ensure_ascii=False, indent=2) + "\n",
        encoding="utf-8",
    )


def write_rows(filename, rows):
    """검토 기록을 한 줄에 한 항목인 JSONL로 저장한다."""
    content = "\n".join(json.dumps(row, ensure_ascii=False) for row in rows)
    (output_dir / filename).write_text(content + "\n", encoding="utf-8")


def triple_key(row):
    """트리플 비교용 (주어, 관계, 목적어) 튜플을 반환한다."""
    # 평가 단계에서는 이름을 정규화하지 않고 원본 표기를 그대로 사용
    return (row["subject"], row["relation"], row["object"])
```

### 관계·타입 스키마 검사

관계별로 허용되는 **주어 타입과 목적어 타입**을 정의하고, 추출 결과를 통과·기각으로 나눈다.

```python
signatures = {
    "TREATS": ("Compound", "Disease"),
    "PALLIATES": ("Compound", "Disease"),
    "BINDS": ("Compound", "Gene"),
    "UPREGULATES_CG": ("Compound", "Gene"),
    "DOWNREGULATES_CG": ("Compound", "Gene"),
    "ASSOCIATES": ("Disease", "Gene"),
    "PRESENTS": ("Disease", "Symptom"),
    "INCLUDES": ("PharmacologicClass", "Compound"),
}


def check_signature(row):
    """스키마 위반 사유를 반환하고, 통과하면 None을 반환한다."""
    if row["relation"] not in signatures:
        return "허용 관계가 아님"

    subject_type, object_type = signatures[row["relation"]]

    if row["subject_type"] != subject_type:
        return f"주어 타입이 {subject_type} 이어야 함"

    if row["object_type"] != object_type:
        return f"목적어 타입이 {object_type} 이어야 함"

    return None


def split_schema(rows):
    """추출 목록을 스키마 통과 목록과 기각 목록으로 나눈다."""
    valid, rejected = [], []

    for row in rows:
        reason = check_signature(row)

        if reason is None:
            valid.append(row)
        else:
            # 원본은 유지하고 기각 목록에만 사유 추가
            rejected.append(dict(row, reject_reason=reason))

    return valid, rejected
```

### 완전일치 평가 함수

중복 트리플은 집합으로 합친 뒤 **TP, FP, FN, Precision, Recall, F1**을 계산한다.

```python
def measure_exact(rows, gold_rows):
    """고유 트리플의 완전일치 평가 지표를 반환한다."""
    predicted = {triple_key(row) for row in rows}
    expected = {triple_key(row) for row in gold_rows}

    tp = len(predicted & expected)   # 추출과 골드 양쪽에 존재
    fp = len(predicted - expected)   # 추출에는 있지만 골드에는 없음
    fn = len(expected - predicted)   # 골드에는 있지만 추출하지 못함

    precision = tp / len(predicted) if predicted else None
    recall = tp / len(expected) if expected else None

    # 골드가 있으면 계산. 골드가 없으면 미산출 처리
    f1 = 2 * tp / (2 * tp + fp + fn) if expected else None

    return {
        "predicted": len(predicted),
        "gold": len(expected),
        "tp": tp,
        "fp": fp,
        "fn": fn,
        "precision": precision,
        "recall": recall,
        "f1": f1,
    }
```

## 2. 사람 판정 기반 정밀도

스키마와 근거 검사를 통과한 트리플을 사람이 다시 보고 `1` 또는 `0`으로 판정.

- `1`: 근거가 관계를 뒷받침하고 개체 표기도 허용 가능
- `0`: 관계가 근거로 충분히 뒷받침되지 않거나 판정 기준을 만족하지 않음

실제 실행에서는 평가 대상 전체를 포함하는 `scores` 딕셔너리가 필요함.

```python
# 형태 예시
scores = {
    ("Crohn's disease", "ASSOCIATES", "NOD2"): 1,
    ("Crohn's disease", "ASSOCIATES", "ATG16L1"): 1,
    # ... 평가 대상 전체 트리플의 0/1 판정값
}
```

### 2-1. 미채점 항목을 차단하고 전수 정밀도 계산

미채점 트리플이 섞인 상태에서 평균을 계산하지 않도록 먼저 검사한다.

```python
def judged_precision(rows, score_table):
    """모든 트리플이 채점됐는지 확인한 뒤 0/1 평균을 계산한다."""
    input_keys = {triple_key(row) for row in rows}
    missing = input_keys - set(score_table)

    if missing:
        raise ValueError(f"미채점 트리플이 있습니다: {missing}")

    if not rows:
        return None

    row_scores = [score_table[triple_key(row)] for row in rows]
    return sum(row_scores) / len(rows)


# 스키마 + 근거 검사를 모두 통과한 22행 전수 평가
precision_all = judged_precision(evidence_ok, scores)
print("precision_all:", precision_all)
```

**결과**

```text

```

전수 판정 정밀도: 약 **72.73%**.

### 3-4. TP, FP, FN 계산

원본 코드는 문장 12만 범위 검사에 사용했지만, 평가 범위는 `[12, 13]`이므로 두 문장을 모두 검사하도록 보완.

```python
# 평가 문장 원문을 미리 저장
mini_sentences = {
    sent_id: docs[mini_doc_id]["sentences"][sent_id]
    for sent_id in mini_plan["sent_ids"]
}

# 해당 문서의 ASSOCIATES 추출 결과
mini_predictions = [
    row for row in valid
    if row["source_doc_id"] == mini_doc_id
    and row["relation"] == "ASSOCIATES"
]


def in_annotation_scope(row):
    """주어와 목적어가 평가 범위의 같은 문장에 함께 있는지 확인한다."""
    return any(
        row["subject"] in mini_sentences[sent_id]
        and row["object"] in mini_sentences[sent_id]
        for sent_id in mini_plan["sent_ids"]
    )


mini_scope_ok = all(
    in_annotation_scope(row) for row in mini_predictions
)

mini_pred_keys = {triple_key(row) for row in mini_predictions}
mini_gold_keys = {triple_key(row) for row in mini_gold}

mini_tp = len(mini_pred_keys & mini_gold_keys)
mini_fp = len(mini_pred_keys - mini_gold_keys)
mini_fn = len(mini_gold_keys - mini_pred_keys)

print("범위 확인:", mini_scope_ok)
print("TP:", mini_tp)
print("FP:", mini_fp)
print("FN:", mini_fn)
```

**결과**

```text

TP: 2
FP: 0
FN: 0
```

### 3-5. Precision, Recall, F1 계산

```python
def prf(tp, fp, fn):
    precision = tp / (tp + fp)              # 추출한 것 중 맞은 비율
    recall = tp / (tp + fn)                 # 골드 중 찾아낸 비율
    f1 = 2 * tp / (2 * tp + fp + fn)        # 정밀도와 재현율을 함께 반영
    return (precision, recall, f1)


mini_prf = prf(mini_tp, mini_fp, mini_fn)
mini_metrics = measure_exact(mini_predictions, mini_gold)

print("직접 계산:", mini_prf)
print(
    "제공 함수:",
    (
        mini_metrics["precision"],
        mini_metrics["recall"],
        mini_metrics["f1"],
    ),
)

print(
    "두 결과가 같은가:",
    mini_prf
    == (
        mini_metrics["precision"],
        mini_metrics["recall"],
        mini_metrics["f1"],
    ),
)
```

**결과**

```text
직접 계산: (1.0, 1.0, 1.0)
제공 함수: (1.0, 1.0, 1.0)
두 결과가 같은가: True
```

미니 골드 범위에서는 **Precision = Recall = F1 = 1.0**.

    scores[triple_key(row)] == 0
    for row in sample

report = {
