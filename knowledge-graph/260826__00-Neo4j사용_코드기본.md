# 1단계: 기초

## 기본 연결 활성화

```python
# Neo4j 연결: 이 셀은 실행만 하세요(내용은 이해하지 않아도 됩니다).
# - .env의 NEO4J_URI / NEO4J_USER / NEO4J_PASSWORD로 데이터베이스에 연결합니다.
# - run_cypher("쿼리", 파라미터=값)가 결과를 dict 리스트로 돌려줍니다.
# - 이 단원은 그래프를 조회만 합니다(그래프를 바꾸지 않습니다).

import os

from dotenv import load_dotenv
from neo4j import GraphDatabase  # 파이썬용 공식 드라이버. 이 클래스로 접속 통로를 연다


# 1) 접속 정보 읽기
# .env에 적힌 값을 환경변수로 올린다.
# 파일이 없으면 조용히 넘어간다.
load_dotenv(".env")       # 같은 폴더의 .env
load_dotenv("../.env")    # 정답 폴더에서 실행하는 경우


# 두 번째 인자는 .env에 해당 키가 없을 때 쓰는 기본값.
# .env를 못 읽어도 에러가 아니라 이 값으로 넘어가므로,
# 마지막에 출력되는 주소가 실습용 DB가 맞는지 확인한다.
NEO4J_URI = os.getenv("NEO4J_URI", "bolt://localhost:7687")
NEO4J_USER = os.getenv("NEO4J_USER", "neo4j")
NEO4J_PASSWORD = os.getenv("NEO4J_PASSWORD", "neo4j")


# 2) 드라이버 생성
# 접속 통로 하나를 노트북 전체에서 나눠 쓴다.
driver = GraphDatabase.driver(
    NEO4J_URI,
    auth=(NEO4J_USER, NEO4J_PASSWORD)
)

# 실제로 연결되는지 확인
driver.verify_connectivity()


# 3) 수업 내내 사용하는 헬퍼
# 쿼리를 보내고 결과를 파이썬 자료형으로 변환한다.
def run_cypher(query, **params):
    """Cypher 실행 -> 결과를 dict 리스트로 반환."""
    with driver.session() as session:
        # with 블록을 벗어나면 세션이 자동으로 닫힌다.
        # record.data()는 결과 한 행을 dict로 바꾼다.
        return [
            record.data()
            for record in session.run(query, **params)
        ]


print("Neo4j 연결:", NEO4J_URI)
```

## 한 쌍을 잇는 관계가 3종 이상인 경우 찾기

먼저 관계가 **2종 이상**인 사람-영화 쌍을 조회한 뒤, Python에서 관계가 **3종 이상**인 경우만 추출.

```python
# 한 쌍을 잇는 관계가 2종 이상인 (사람, 영화) 쌍
multi_rows = run_cypher(
    "MATCH (p:Person)-[r]->(m:Movie) "
    "WITH p, m, collect(DISTINCT type(r)) AS types "
    "WHERE size(types) >= 2 "
    "RETURN p.name AS person, m.title AS movie, types AS types "
    "ORDER BY p.name"
)

# 일부 확인
for row in multi_rows[:3]:
    print(row)


# 관계가 3종 이상인 쌍만 추출
triples = []

for row in multi_rows:
    if len(row["types"]) >= 3:
        triples.append(
            (row["person"], row["movie"])
        )


print("---")

n_triple = len(triples)

print(triples)
print(n_triple)
```

## 출연 관계를 이용해 다작 배우 찾기

```python
# 모든 (배우, 영화) 출연 쌍 가져오기
actor_pairs = run_cypher(
    "MATCH (p:Person)-[:ACTED_IN]->(m:Movie) "
    "RETURN p.name AS actor, m.title AS movie"
)

print(actor_pairs[:5])


# 배우 이름만 모으기
actor_names = []

for pair in actor_pairs:
    actor_names.append(pair["actor"])

actor_counts = Counter(actor_names)


# 가장 많이 출연한 배우 1명
top_actor = actor_counts.most_common(1)[0]
```

## 관계 속성을 이용해 최고 평점 영화 찾기

`REVIEWED` 관계의 `rating` 속성을 가져온 뒤 DataFrame에서 가장 높은 값을 찾음.

```python
# 리뷰(리뷰어·영화·점수)를 DataFrame으로 가져오기
reviews_df = pd.DataFrame(
    run_cypher(
        "MATCH (p:Person)-[r:REVIEWED]->(m:Movie) "
        "RETURN p.name AS reviewer, "
        "m.title AS movie, "
        "r.rating AS rating"
    )
)

display(reviews_df[:3])


# 최고 평점이 있는 행의 인덱스 찾기
best_idx = reviews_df["rating"].idxmax()


# 최고 평점 행 가져오기
best_row = reviews_df.loc[best_idx]


# 영화 제목과 평점 저장
best_movie = best_row["movie"]
best_rating = int(best_row["rating"])

print("영화 제목:", best_movie)
print("평점:", best_rating)
```
