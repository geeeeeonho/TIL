# 문서에서 그래프 생성 및 품질 검증

> **핵심 흐름**  
> 원문 확인 → 스키마 정의 → 청크 분할 → LLM 관계 추출 → Neo4j 저장 → 평가용 결과 저장 → Entity Resolution → 스키마·근거·Gold 평가 → 청크 조건 변경 비교

의약품 설명서에서 `Drug - HAS_SIDE_EFFECT -> Symptom` 관계를 추출하고, 저장한 그래프의 품질까지 검증하는 전체 흐름이다.

## 0. 공통 환경 준비

**핵심 설명**  
파일 입출력, Neo4j 연결, LLM·임베딩 모델을 한 번만 준비한다.

### 코드

```python
import json
import os
from pathlib import Path
from copy import deepcopy
from functools import partial
from uuid import uuid4, UUID

from dotenv import find_dotenv, load_dotenv
from neo4j import GraphDatabase
from neo4j_graphrag.llm import OpenAILLM
from neo4j_graphrag.embeddings import OpenAIEmbeddings
from langchain_text_splitters import RecursiveCharacterTextSplitter
from neo4j_graphrag.experimental.components.text_splitters.langchain import (
    LangChainTextSplitterAdapter,
)
from neo4j_graphrag.experimental.pipeline.kg_builder import SimpleKGPipeline
from neo4j_graphrag.generation.prompts import ERExtractionTemplate
from neo4j_graphrag.experimental.components.resolver import (
    SinglePropertyExactMatchResolver,
)

material_dir = Path(".")
data_dir = material_dir / "data"
output_dir = material_dir / "output"
output_dir.mkdir(exist_ok=True)


def read_json(path):
    return json.loads(path.read_text(encoding="utf-8"))


def save_json(path, value):
    path.write_text(
        json.dumps(value, ensure_ascii=False, indent=2),
        encoding="utf-8",
    )


load_dotenv(find_dotenv(usecwd=True))

driver = GraphDatabase.driver(
    os.environ["NEO4J_URI"],
    auth=(os.environ["NEO4J_USER"], os.environ["NEO4J_PASSWORD"]),
)
driver.verify_connectivity()


def run_cypher(query, **params):
    with driver.session() as session:
        return [record.data() for record in session.run(query, **params)]


llm = OpenAILLM(model_name="gpt-5.6-luna")
embedder = OpenAIEmbeddings(model="text-embedding-3-large")
embedder.embed_query = partial(embedder.embed_query, dimensions=768)

print("Neo4j 연결 완료")
```

## 2. 청크 분할과 임베딩

**핵심 설명**  
기본 실험은 `chunk_size=300`, `chunk_overlap=80`을 사용한다. 제품명과 `[이상반응]` 절이 같은 청크에 있어야 관계 추출에 필요한 문맥이 확보된다.

```python

    "chunk_overlap": 80,

task_text_splitter = RecursiveCharacterTextSplitter(**task_split_settings)

task_chunks = await task_splitter.run(task_doc["text"])

task_vector = embedder.embed_query(task_chunks.chunks[0].text)

print("청크 수:", len(task_chunks.chunks))

```

- 청크 1: `[사용법]` + `[주의사항]` + `[상호작용]`

## 3. LLM 관계 추출 및 Neo4j 저장

**핵심 설명**  
스키마와 추출 규칙을 `SimpleKGPipeline`에 연결한다. 평가 전 원본을 보존하기 위해 자동 Entity Resolution은 끈다.

### 코드

```python
prompt_template = (
    ERExtractionTemplate.DEFAULT_TEMPLATE
    + """
Drug - HAS_SIDE_EFFECT -> Symptom 관계만 추출하세요.
청크에 제품명과 [이상반응] 절이 모두 있을 때만 추출하세요.

[이상반응] 절의 각 증상은 개별 관계로 기록하세요.
효능, 사용법, 주의사항, 상호작용의 언급은 제외하세요.
evidence에는 이상반응을 안내하는 원문 문장 전체를 그대로 인용하세요.
"""
)

builder = SimpleKGPipeline(
    llm=llm,
    driver=driver,
    embedder=embedder,
    schema=deepcopy(schema),
    prompt_template=prompt_template,
    text_splitter=task_splitter,
    from_file=False,
    on_error="RAISE",
    perform_entity_resolution=False,
)

execution_id = str(uuid4())

result = await builder.run_async(
    text=task_doc["text"],
    file_path=task_doc["url"],
    document_metadata={
        "source_doc_id": task_doc["doc_id"],
        "execution_id": execution_id,
    },
)

writer = result.result["writer"]
print("writer 상태:", writer["status"])
print(writer["metadata"]["statistics"])
```

## 4. 저장 관계 조회와 Baseline 저장

**핵심 설명**  
현재 실행에서 생성한 관계와 청크만 조회하고, **Entity Resolution 전 결과**를 평가 기준으로 저장한다.

### 코드

```python
def read_relations(execution_id):
    return run_cypher(
        """
        MATCH (d:Document {execution_id: $execution_id})
              <-[:FROM_DOCUMENT]-(c:Chunk)
              <-[:FROM_CHUNK]-(s:__Entity__)
        MATCH (s)-[r]->(o:__Entity__)-[:FROM_CHUNK]->(c)

        RETURN
            s.name AS subject,
            head([x IN labels(s) WHERE NOT x STARTS WITH '__']) AS subject_type,
            type(r) AS relation,
            o.name AS object,
            head([x IN labels(o) WHERE NOT x STARTS WITH '__']) AS object_type,
            coalesce(r.evidence, '') AS evidence,
            d.source_doc_id AS source_doc_id,
            collect(DISTINCT c.text) AS chunk_texts

        ORDER BY subject, relation, object
        """,
        execution_id=execution_id,
    )


def read_chunks(execution_id):
    return run_cypher(
        """
        MATCH (c:Chunk)-[:FROM_DOCUMENT]->
              (d:Document {execution_id: $execution_id})
        RETURN c.index AS index,
               c.text AS text,
               size(c.embedding) AS dimensions
        ORDER BY index
        """,
        execution_id=execution_id,
    )


task_rows = read_relations(execution_id)
task_stored_chunks = read_chunks(execution_id)

for row in task_rows:
    print(row["subject"], "->", row["relation"], "->", row["object"])

baseline = {
    "execution_id": execution_id,
    "document": task_doc,
    "chunk_size": 300,
    "chunk_overlap": 80,
    "schema": schema,
    "prompt_template": prompt_template,
    "perform_entity_resolution": False,
    "rows": task_rows,
    "chunks": task_stored_chunks,
}

save_json(output_dir / "baseline_assignment.json", baseline)
```

### 결과

```text
이노엔비타메진캡슐 -> HAS_SIDE_EFFECT -> 구역
이노엔비타메진캡슐 -> HAS_SIDE_EFFECT -> 구토
이노엔비타메진캡슐 -> HAS_SIDE_EFFECT -> 묽은 변
이노엔비타메진캡슐 -> HAS_SIDE_EFFECT -> 복부팽만감
이노엔비타메진캡슐 -> HAS_SIDE_EFFECT -> 식욕부진

저장 파일: output/baseline_assignment.json
```

## 5. Entity Resolution

**핵심 설명**  
같은 실행 안에서 **레이블과 이름이 정확히 같은 노드**를 통합한다. 동의어나 별칭을 의미적으로 판단하는 정규화와는 다르다.

### 코드

```python
async def merge_this_execution(execution_id):
    checked_id = str(UUID(execution_id))

    filter_query = f"""
    WHERE EXISTS {{
        MATCH (entity)-[:FROM_CHUNK]->(:Chunk)-[:FROM_DOCUMENT]->(d:Document)
        WHERE d.execution_id = '{checked_id}'
    }}
    """

    resolver = SinglePropertyExactMatchResolver(
        driver=driver,
        filter_query=filter_query,
    )

    return await resolver.run()


resolution = await merge_this_execution(execution_id)

print("통합 대상:", resolution.number_of_nodes_to_resolve)
print("통합 후:", resolution.number_of_created_nodes)
```

### 결과

```text
통합 대상: 6
통합 후: 6
```

이번 실행에는 정확히 같은 이름·레이블의 중복 노드가 없었다.

# 그래프 품질 검증

## 6. 스키마 준수와 근거 원문 일치 검사

**핵심 설명**  
스키마 검사는 **구조**, 근거 검사는 **인용 문장의 존재 여부**를 본다. 둘은 별도 평가다.

### 코드

```python
def schema_ok(row):
    return (
        row["subject_type"],
        row["relation"],
        row["object_type"],
    ) == (
        "Drug",
        "HAS_SIDE_EFFECT",
        "Symptom",
    )


def evidence_ok(row, document):
    return (
        row["source_doc_id"] == document["doc_id"]
        and isinstance(row["evidence"], str)
        and bool(row["evidence"].strip())
        and row["evidence"] in document["text"]
    )


def triple_key(row):
    return (
        row["subject"],
        row["relation"],
        row["object"],
    )


task_snapshot = read_json(output_dir / "baseline_assignment.json")
task_rows = task_snapshot["rows"]
task_doc = task_snapshot["document"]

total = len(task_rows)
schema_count = sum(schema_ok(row) for row in task_rows)
evidence_count = sum(evidence_ok(row, task_doc) for row in task_rows)

print(f"스키마 준수율: {schema_count}/{total} ({schema_count / total:.1%})")
print(f"근거 원문 일치율: {evidence_count}/{total} ({evidence_count / total:.1%})")
```

### 결과

```text
스키마 준수율: 5/5 (100.0%)
근거 원문 일치율: 5/5 (100.0%)
```

> `evidence_ok=True`는 인용 문장이 실제 원문에 있다는 뜻이다. 관계 자체가 정답인지는 Gold 평가가 따로 필요하다.

## 7. Gold 기준 Precision · Recall · F1

**핵심 설명**  
중복 저장 가능성을 제거하기 위해 `(주어, 관계, 목적어)`를 집합으로 만든 뒤 Gold와 비교한다.

### 코드

```python
def score_relations(rows, gold):
    predicted = {triple_key(row) for row in rows}
    expected = {triple_key(row) for row in gold}

    tp = predicted & expected
    fp = predicted - expected
    fn = expected - predicted

    precision = len(tp) / len(predicted) if predicted else None
    recall = len(tp) / len(expected) if expected else None

    denominator = 2 * len(tp) + len(fp) + len(fn)
    f1 = 2 * len(tp) / denominator if denominator else None

    return {
        "tp": tp,
        "fp": fp,
        "fn": fn,
        "precision": precision,
        "recall": recall,
        "f1": f1,
    }


task_gold = read_json(data_dir / "assignment_gold.json")
metrics = score_relations(task_rows, task_gold)

print("TP:", len(metrics["tp"]))
print("FP:", len(metrics["fp"]))
print("FN:", len(metrics["fn"]))
print("Precision:", metrics["precision"])
print("Recall:", metrics["recall"])
print("F1:", metrics["f1"])
```

### 결과

```text
원본 LV2 노트북에는 실제 실행 출력이 저장되어 있지 않음.

실행 후 확인:
- TP / FP / FN
- Precision / Recall / F1
- 실제 FP / FN 관계
```

평가 중 실패 행이나 Gold 항목을 임의로 제거하면 비교 기준이 달라지므로 그대로 유지한다.

## 8. 청크 크기 변경 실험: 300 → 1000

large_text_splitter = RecursiveCharacterTextSplitter(**large_split_settings)

large_chunks = await large_splitter.run(task_doc["text"])

large_builder = SimpleKGPipeline(
    llm=llm,

    embedder=embedder,
    schema=deepcopy(schema),
    prompt_template=prompt_template,

large_execution_id = str(uuid4())

large_result = await large_builder.run_async(
    text=task_doc["text"],
    file_path=task_doc["url"],
    document_metadata={
        "source_doc_id": task_doc["doc_id"],
        "execution_id": large_execution_id,
    },
)

large_rows = read_relations(large_execution_id)
large_stored_chunks = read_chunks(large_execution_id)

large_snapshot = {
    "execution_id": large_execution_id,
    "document": task_doc,
    "chunk_size": 1000,
    "chunk_overlap": 80,
    "schema": schema,
    "prompt_template": prompt_template,
    "perform_entity_resolution": False,

```text

저장 파일: output/large_assignment.json
```

원문 길이가 약 493자이므로 `chunk_size=1000`에서는 전체 문서가 하나의 청크에 들어간다.

실제 관계 수와 점수는 LLM 재실행 결과에 따라 달라질 수 있다.

## 9. 변경 전·후 동일 기준 비교

**핵심 설명**  
300자와 1000자 결과에 같은 Gold와 같은 평가 함수를 적용한다. 점수뿐 아니라 **FP/FN이 왜 달라졌는지** 확인하는 것이 핵심이다.

### 코드

```python
baseline = read_json(output_dir / "baseline_assignment.json")
large = read_json(output_dir / "large_assignment.json")

gold = read_json(data_dir / "assignment_gold.json")

before = score_relations(baseline["rows"], gold)
after = score_relations(large["rows"], gold)

for label, snapshot, result in [
    ("변경 전", baseline, before),
    ("변경 후", large, after),
]:
    rows = snapshot["rows"]
    total = len(rows)

    print(f"=== {label} ===")
    print("chunk_size:", snapshot["chunk_size"])
    print("청크 수:", len(snapshot["chunks"]))
    print("스키마 준수:", sum(schema_ok(row) for row in rows), "/", total)
    print("근거 일치:", sum(evidence_ok(row, snapshot["document"]) for row in rows), "/", total)
    print("Precision:", result["precision"])
    print("Recall:", result["recall"])
    print("F1:", result["f1"])
    print("FP:", result["fp"])
    print("FN:", result["fn"])
    print()

driver.close()
```

### 결과

```text
확정 가능한 변화
- chunk_size: 300 → 1000
- chunk_overlap: 80 유지
- 청크 수: 2 → 1
- 원문 / 모델 / 스키마 / 프롬프트 / Gold 유지

Precision / Recall / F1 / FP / FN의 실제 변경값은 재실행 후 확인
```

# 전체 구조 요약

```text
원문
 ↓
스키마 정의
 ↓
청크 분할 + 임베딩
 ↓
LLM 관계 추출
 ↓
Neo4j 저장
 ↓
baseline_assignment.json
 ↓
┌───────────────────────┬────────────────────────┐
│ Entity Resolution     │ 품질 평가              │
│ 이름+레이블 정확 일치 │ 스키마 / 근거 / Gold   │
└───────────────────────┴────────────────────────┘
                          ↓
                   chunk_size 변경
                    300 → 1000
                          ↓
                     재추출·재평가
```

## 핵심 포인트

- **스키마**: 추출 가능한 노드·관계 구조 제한
- **청크**: 필요한 문맥이 같은 청크에 있는지가 중요
- **evidence**: 실제 원문 근거를 관계에 보존
- **Baseline**: Entity Resolution 전 결과를 평가 기준으로 저장
- **Entity Resolution**: 동일 이름·동일 레이블 노드의 정확 일치 통합
- **스키마 검사**: 관계 구조 검증
- **근거 검사**: evidence가 원문에 실제 존재하는지 검증
- **Gold 평가**: 관계 자체의 정답 여부를 Precision / Recall / F1로 평가
- **비교 실험**: 한 조건만 바꾸고 나머지는 동일하게 유지
- **해석**: 최종 점수만 보지 말고 FP/FN 변화 원인을 확인
