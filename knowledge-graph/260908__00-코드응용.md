# 트리플 추출 · 진단 · 정제 · 그래프 적재 응용

> 전체 흐름: **온톨로지 정의 → 배치 진단 → 시그니처 검사 → 중복 제거 → 이름-ID 정규화 → Candidate 격리 → Cypher MERGE → LLM 실시간 추출/검증**

## 0. 기본 준비

### 이름 → ID 사전 준비

추출된 이름을 기존 지식그래프의 정식 ID와 연결하기 위한 준비 단계다.

```python
import json
import re
from pathlib import Path

NAME2ID = json.loads(
    Path("data/name2id.json").read_text(encoding="utf-8")
)


def normalize_name(name):
    """사전 조회용 이름 정규화."""
    s = re.sub(r"\s+", " ", name).strip().lower()
    return re.sub(r"\s*\([^)]*\)$", "", s).strip()


def lookup_id(name, node_type):
    """이름과 타입으로 Hetionet ID를 찾아 (id, reason)을 반환."""

    # Gene symbol은 대소문자가 의미를 가질 수 있으므로 그대로 조회
    if node_type == "Gene":
        found = NAME2ID["genes"].get(name.strip())
        return (found, "hit") if found else (None, "miss")

    key = normalize_name(name)

    # 하나의 이름이 여러 타입에 걸리면 자동 확정하지 않음
    if key in NAME2ID["ambiguous"]:
        return None, "ambiguous"

    entry = NAME2ID["entries"].get(key)

    # 이름뿐 아니라 타입도 일치해야 연결
    if entry and entry["label"] == node_type:
        return entry["id"], "hit"

    return None, "miss"


print(
    "사전 항목:", len(NAME2ID["entries"]),
    "/ 유전자 기호:", len(NAME2ID["genes"]),
    "/ 애매한 이름:", len(NAME2ID["ambiguous"])
)
```

**결과**

```text
사전 항목: 5725 / 유전자 기호: 13113 / 애매한 이름: 9
```

### 구조화 출력 + 추출 프롬프트 준비

Pydantic으로 모델 출력 형식을 제한하고, 온톨로지 관계를 프롬프트에 자동으로 넣는다.

```python
from typing import Literal
from pydantic import BaseModel, Field
from langchain_core.prompts import ChatPromptTemplate

# 현재 온톨로지에 정의된 값만 허용
RelationName = Literal[*tuple(RELATION_SIGNATURES.keys())]
NodeType = Literal[*tuple(sorted(NODE_TYPES))]


class Triple(BaseModel):
    subject: str = Field(description="주어. 논문 표기 그대로")
    subject_type: NodeType = Field(description="주어 타입")
    relation: RelationName = Field(description="허용된 관계 중 하나")
    object: str = Field(description="목적어. 논문 표기 그대로")
    object_type: NodeType = Field(description="목적어 타입")
    evidence: str = Field(
        description="트리플을 직접 뒷받침하는 원문 문장 하나",
        max_length=200,
    )


class Extraction(BaseModel):
    triples: list[Triple] = Field(description="문서에서 추출한 트리플 목록")


def build_ontology_block(sigs):
    """관계 시그니처를 프롬프트용 텍스트로 변환."""
    lines = ["[허용 관계]"]

    for rel, (subj, obj, criterion) in sigs.items():
        lines.append(
            f"- {rel}: ({subj}) -> ({obj}) # {criterion}"
        )

    return "\n".join(lines)


def build_extraction_prompt():
    ontology = build_ontology_block(RELATION_SIGNATURES)

    system = (
        "너는 의학 논문에서 지식 트리플을 추출하는 도구야.\n\n"
        f"{ontology}\n\n"
        "[규칙]\n"
        "- 허용된 관계만 사용하고 주어 타입 -> 목적어 타입 방향을 지킨다.\n"
        "- 맞는 관계가 없으면 억지로 분류하지 않고 제외한다.\n"
        "- 주어와 목적어에는 개체 이름만 넣고 논문 표기를 유지한다.\n"
        "- evidence에는 해당 사실을 직접 뒷받침하는 원문 문장 하나만 넣는다."
    )

    return ChatPromptTemplate.from_messages([
        ("system", system),
        ("human", "[문서]\n{text}"),
    ])


LIVE_TEXT = Path("data/pgx_lv2_corpus.txt").read_text(encoding="utf-8")

extraction_chain = (
    build_extraction_prompt()
    | make_model().with_structured_output(Extraction)
)

print("라이브 추출용 발췌:", len(LIVE_TEXT), "자")
print(LIVE_TEXT[:120], "...")
```

**결과**

```text
라이브 추출용 발췌: 1801 자
Introduction
Methotrexate (MTX) suppresses immune response and reduces vaccine efficacy in patients with rheumatoid art ...
```

> `RelationName`과 `NodeType`은 스키마를 만들 때 값이 고정됨. 온톨로지를 수정했다면 구조화 출력 모델도 다시 만들어야 함.

## 1-2. 문서별 트리플 수와 허용 관계 비율

문서별로 전체 트리플 수와 **허용된 관계 이름이 차지하는 비율**을 계산한다.

```python
# 문서별 전체 트리플 수
by_doc = {}

# 문서별 허용 관계 수
allowed_by_doc = {}

for triple in batch:
    doc_id = triple["doc_id"]
    relation = triple["relation"]

    by_doc[doc_id] = by_doc.get(doc_id, 0) + 1

    if relation in RELATION_SIGNATURES:
        allowed_by_doc[doc_id] = allowed_by_doc.get(doc_id, 0) + 1


# 문서별 허용 관계 비율
allowed_ratio = {}

for doc_id, total in by_doc.items():
    allowed = allowed_by_doc.get(doc_id, 0)
    allowed_ratio[doc_id] = round(allowed / total, 2)


print("문서별 전체 트리플 수:", by_doc)
print("문서별 허용 관계 비율:", allowed_ratio)
print(by_doc["PMC13495420"])

print(allowed_ratio["PMC13493261"])
```

# 2. 정제와 그래프 적재 준비

## 2-1. 허용 관계만 남기고 중복 제거

관계 이름이 온톨로지에 있는 트리플만 남긴 뒤 동일 트리플을 하나로 합친다.

```python
# 1. 허용 관계만 남기기
allowed = [
    row
    for row in batch
    if row["relation"] in RELATION_SIGNATURES
]


# 2. 타입 정보를 유지하면서 중복 제거
seen = set()
clean = []

for row in allowed:
    key = (
        row["subject"],
        row["subject_type"],
        row["relation"],
        row["object"],
        row["object_type"],
    )

    if key not in seen:
        seen.add(key)
        clean.append(row)


print("허용 관계 트리플 수:", len(allowed))
print("중복 제거 후:", len(clean))

for row in clean:
    print((row["subject"], row["relation"], row["object"]))
```

**결과**

```text

('Belimumab', 'BINDS', 'BAFF')
```

## 2-2. 관계 시그니처로 타입과 방향 검사

관계 이름뿐 아니라 **주어 타입과 목적어 타입의 방향까지** 확인한다.

```python
def signature_ok(row):
    relation = row["relation"]

    # 허용되지 않은 관계
    if relation not in RELATION_SIGNATURES:
        return False

    expected_subject_type, expected_object_type, _ = \
        RELATION_SIGNATURES[relation]

    return (
        row["subject_type"] == expected_subject_type
        and row["object_type"] == expected_object_type
    )


bad = [
    row
    for row in batch
    if not signature_ok(row)
]


# 방향을 일부러 뒤집은 테스트
flipped = {
    "subject": "fatigue",
    "subject_type": "Symptom",
    "relation": "PRESENTS",
    "object": "sarcoidosis",
    "object_type": "Disease",
}


print("잘못된 트리플 수:", len(bad))
print("잘못된 트리플:", bad)
print("뒤집은 트리플:", signature_ok(flipped))
```

**결과**

```text
잘못된 트리플 수: 1
잘못된 트리플:
[{'subject': 'prednisone', 'subject_type': 'Compound',
  'relation': 'CAUSES', 'object': 'hypertension',
  'object_type': 'Disease', ...}]

뒤집은 트리플: False
```

## 2-3. 이름 → ID 연결 + 미해결 개체 분리

기존 그래프에 연결되는 개체는 ID를 붙이고, 찾지 못한 이름은 Candidate 후보로 분리한다.

```python
entities = set()

for row in clean:
    entities.add((row["subject"], row["subject_type"]))
    entities.add((row["object"], row["object_type"]))


resolved = {}
candidate_names = []

for name, entity_type in entities:
    entity_id, reason = lookup_id(name, entity_type)

    if entity_id is not None:
        resolved[name] = entity_id
    else:
        candidate_names.append(name)


candidates = sorted(candidate_names)

print("resolved:")
print(resolved)

print("\ncandidates:")
print(candidates)

print("\nmethotrexate:", resolved["methotrexate"])
print("Belimumab 후보 여부:", "Belimumab" in candidates)
```

## 2-4. Cypher MERGE 문 생성

ID가 확인된 개체는 기존 노드에 연결하고, 미해결 개체는 `Candidate`로 격리한다.

```python
def node_pattern(alias, name, node_type):
    # 기존 그래프의 정식 노드
    if name in resolved:
        node_id = resolved[name]
        return f"({alias}:{node_type} {{id: '{node_id}'}})"

    # 아직 ID를 확정하지 못한 개체
    return (
        f"({alias}:Candidate "
        f"{{name: '{name}', type: '{node_type}'}})"
    )


def make_merge(row):
    subject_pattern = node_pattern(
        "a", row["subject"], row["subject_type"]
    )

    object_pattern = node_pattern(
        "b", row["object"], row["object_type"]
    )

    relation = row["relation"]

    return (
        f"MERGE {subject_pattern}\n"
        f"MERGE {object_pattern}\n"
        f"MERGE (a)-[:{relation}]->(b)"
    )


# 중복 제거된 트리플을 사용
merges = [
    make_merge(row)
    for row in clean
    if signature_ok(row)
]


print("MERGE 문 개수:", len(merges))

for merge in merges:
    print(merge)
    print()
```

**결과**

```text
MERGE 문 개수: 11

MERGE (a:Candidate {name: 'Belimumab', type: 'Compound'})
MERGE (b:Candidate {name: 'BAFF', type: 'Gene'})
MERGE (a)-[:BINDS]->(b)

MERGE (a:Candidate {name: 'Belimumab', type: 'Compound'})
MERGE (b:Candidate {name: 'SLE', type: 'Disease'})
MERGE (a)-[:TREATS]->(b)

MERGE (a:Compound {id: 'Compound::DB00635'})
MERGE (b:Candidate {name: 'SLE', type: 'Disease'})
MERGE (a)-[:TREATS]->(b)

...
```

> 예제 이해를 위해 문자열로 Cypher를 만듦. 실제 적재에서는 따옴표가 포함된 이름과 보안을 고려해 **파라미터 쿼리**를 쓰는 편이 안전함.

## 2-5. 타입 계층을 보조 레이블로 확장

세부 타입이 있을 때 상위 타입 레이블도 함께 부여하는 간단한 방법이다.

```python
TYPE_HIERARCHY = {
    "Cytokine": "Gene",
    "Autoimmune": "Disease",
}


def labels_for(node_type):
    parent = TYPE_HIERARCHY.get(node_type)

    if parent is not None:
        return [parent, node_type]

    return [node_type]


def label_clause(node_type):
    return ":".join(labels_for(node_type))


print(labels_for("Cytokine"))
print(labels_for("Compound"))
print(label_clause("Cytokine"))
print(label_clause("Compound"))
```

**결과**

```text
['Gene', 'Cytokine']
['Compound']
Gene:Cytokine
Compound
```

이 코드는 **그래프 적재 시 레이블을 확장하는 예시**임. LLM이 `Cytokine` 같은 세부 타입을 직접 출력하게 하려면 `NODE_TYPES`와 구조화 출력 스키마도 함께 확장해야 함.

# 3. 모델로 직접 추출하기

## 3-1. 실제 추출 + 즉시 정제

구조화 출력 모델로 트리플을 추출하고, 근거·시그니처 검사를 거쳐 최종 트리플만 남긴다.

```python
# 1. 모델 호출
live_result = extraction_chain.invoke({
    "text": LIVE_TEXT
})


# 2. 추출 결과 확인
for tp in live_result.triples:
    print((tp.subject, tp.relation, tp.object))


# 3. evidence가 실제 원문에 있고,
#    주어와 목적어가 evidence 안에 등장하는지 확인
#    -> 문자열 기반의 간단한 1차 검증임
def grounded(tp, text):
    return (
        tp.evidence in text
        and tp.subject in tp.evidence
        and tp.object in tp.evidence
    )


# 4. 관계의 타입 방향까지 확인
def triple_signature_ok(tp):
    expected_subject, expected_object, _ = \
        RELATION_SIGNATURES[tp.relation]

    return (
        tp.subject_type == expected_subject
        and tp.object_type == expected_object
    )


# 5. 검증 + 중복 제거
live_clean = {
    (tp.subject, tp.relation, tp.object)
    for tp in live_result.triples
    if grounded(tp, LIVE_TEXT)
    and triple_signature_ok(tp)
}


print("\n정제 결과:")
print(live_clean)

print("\n정제된 트리플 수:")
print(len(live_clean))
```

**결과**

```text
('MTX', 'BINDS', 'reduced folate carrier 1 (RFC1)')
('MTX', 'BINDS', 'folylpolyglutamate synthase (FPGS)')

정제 결과:
{('MTX', 'BINDS', 'reduced folate carrier 1 (RFC1)'),
 ('MTX', 'BINDS', 'folylpolyglutamate synthase (FPGS)')}

정제된 트리플 수:
2
```

`grounded()`는 **정확한 문자열 포함 여부**만 확인하는 간단한 휴리스틱임. 의미적으로 근거가 충분한지까지 판정하는 검증기는 아님.

## 3-2. 단계형 검증 프롬프트

관계 선택 전에 **개체 → 관계 후보 → 타입 방향 → 최종 트리플** 순서로 확인하도록 지시한다.

```python
from langchain_core.prompts import ChatPromptTemplate


def cot_prompt():
    ontology_block = build_ontology_block(RELATION_SIGNATURES)

    system_text = f"""너는 문서에서 온톨로지에 맞는 지식 트리플을 추출하는 정보 추출기이다.

{ontology_block}

[검증 순서]
1) 문서에서 개체와 개체 타입을 확인한다.
2) 개체 사이의 관계 후보를 찾는다.
3) 주어 타입과 목적어 타입이 관계 시그니처와 맞는지 확인한다.
4) 검사를 통과한 관계만 최종 트리플로 반환한다.
"""

    return ChatPromptTemplate.from_messages([
        ("system", system_text),
        ("human", "[문서]\n{text}"),
    ])


prompt = cot_prompt()
print(prompt.input_variables)

messages = prompt.format_messages(
    text="Methotrexate ..."
)

print("메시지 수:", len(messages))

for message in messages:
    print("-----")
    print(message.content)
```

**결과**

```text
['text']
메시지 수: 2
-----
너는 문서에서 온톨로지에 맞는 지식 트리플을 추출하는 정보 추출기이다.

[허용 관계]
- TREATS: (Compound) -> (Disease) # ...
- PALLIATES: (Compound) -> (Disease) # ...
- BINDS: (Compound) -> (Gene) # ...
...

[검증 순서]
1) 문서에서 개체와 개체 타입을 확인한다.
2) 개체 사이의 관계 후보를 찾는다.
3) 주어 타입과 목적어 타입이 관계 시그니처와 맞는지 확인한다.
4) 검사를 통과한 관계만 최종 트리플로 반환한다.
-----
[문서]
Methotrexate ...
```

`CoT`라는 이름보다 실제 용도는 **단계형 검증 지시**에 가까움. 중간 추론 자체를 결과로 받을 필요는 없고, 최종 구조화 출력만 받으면 됨.

## 핵심 흐름 정리

```text
[LLM / 배치 추출]
        ↓
관계 이름 검사
        ↓
관계 시그니처 검사
(주어 타입 → 목적어 타입)
        ↓
중복 제거
        ↓
이름 → 정식 ID 조회
   ├─ 성공 → 기존 노드 연결
   └─ 실패 → Candidate 격리
        ↓
Cypher MERGE
        ↓
지식그래프 적재
```

핵심은 **모델이 뽑은 값을 바로 그래프에 넣지 않고, 온톨로지·근거·ID 정규화 검사를 거친 뒤 적재하는 것**임.
