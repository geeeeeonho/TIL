## 1. 기본 데이터 불러오기

```python

import json
from pathlib import Path

graph = json.loads(
    Path('data/sns_follows.json').read_text(encoding='utf-8')
)

nodes = graph['nodes']
# id -> {'name': 이름, 'type': 사용자/브랜드, 'region': 지역}

edges = graph['edges']
# [{'relation': 관계, 'source': 출발 id, 'target': 도착 id}, ...]

print('노드', len(nodes), '개 · 관계', len(edges), '개')
print('노드 예:', list(nodes.items())[0])

for e in edges[:3]:
    print(e)
```

### 관계를 트리플로 변환하기

그래프의 `relation`, `source`, `target` 구조를 지식 그래프의 `(주어, 관계, 목적어)` 트리플로 변환.

```python
# 그래프 관계 -> 지식 그래프 트리플로 변환
def edge_to_triple(e):
    # source id를 이름으로 변환
    s = nodes[e['source']]['name']

    # 관계
    r = e['relation']

    # target id를 이름으로 변환
    o = nodes[e['target']]['name']

    # (주어, 관계, 목적어) 형태의 튜플 반환
    return s, r, o


display(edge_to_triple(edges[0]))
```

### 결과

```text
('민준', '팔로우', '서연')
```

### 2홉 관계 찾기

민준이 팔로우하는 사람이 다시 팔로우하는 사용자 찾기.

```python
follow_list = []

# u1이 직접 팔로우하는 사람 찾기
for e in edges:
    if e['relation'] == '팔로우' and e['source'] == 'u1':
        follow_list.append(e['target'])


two_hops = []

# u1이 팔로우하는 사람들이 다시 팔로우하는 사람 찾기
for e in edges:
    if e['relation'] == '팔로우' and e['source'] in follow_list:
        two_hops.append(e['target'])

# 중복 제거
two_hop = set(two_hops)

# 자기 자신 u1은 제외
two_hop = two_hop - {'u1'}

display(two_hop)
```

# SPARQL 실전

## 3. RDF 그래프 생성

기존 SNS 그래프를 `rdflib`의 RDF 그래프로 변환.

```python
# [제공 코드] SNS 그래프를 rdflib 의 RDF 그래프로 옮긴다(실행만 하세요)
from rdflib import Graph, Literal, Namespace, RDF, RDFS

EX = Namespace('http://example.org/sns/')

sns_graph = Graph()
sns_graph.bind('ex', EX)

for node_id, attr in nodes.items():
    # 타입은 표준 술어 rdf:type 으로, 지역은 값(리터럴)으로 담는다
    sns_graph.add((EX[attr['name']], RDF.type, EX[attr['type']]))
    sns_graph.add((EX[attr['name']], EX['지역'], Literal(attr['region'])))

for e in edges:
    sns_graph.add((
        EX[nodes[e['source']]['name']],
        EX[e['relation']],
        EX[nodes[e['target']]['name']]
    ))

# 타입끼리의 계층도 적어 둔다. 사용자도 브랜드도 넓게 보면 '계정'이다
sns_graph.add((EX['사용자'], RDFS.subClassOf, EX['계정']))
sns_graph.add((EX['브랜드'], RDFS.subClassOf, EX['계정']))

# 질의문에서 실제로 쓰는 이름표만 적어 둔다
# rdfs: 는 마지막 계층 문제에서 사용
QUERY_PREFIX = (
    'PREFIX ex: <http://example.org/sns/>\n'
    'PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>\n'
)

print('트리플 수:', len(sns_graph))
```

### 결과

```text
트리플 수: 39
```

## 4. SPARQL 기본 조회

query = QUERY_PREFIX + """

### 타입으로 찾기

### 리터럴 값 FILTER

타입이 `ex:사용자`이면서 `ex:지역` 값이 `부산`인 사람의 이름 집합 조회.

```python
# 쿼리문 작성
query = QUERY_PREFIX + """
SELECT ?person
WHERE {
    # 타입이 사용자인 대상 찾기
    ?person a ex:사용자 .

    # 같은 사람의 지역 값을 ?region에 담기
    ?person ex:지역 ?region .

    # 지역 값이 "부산"인 경우만 남기기
    FILTER(?region = "부산")
}
"""

# 쿼리문 실행, 결과 담기
result = sns_graph.query(query)

# 부산 사용자 이름을 중복 없이 담을 집합 생성
busan_users = set()

# 결과에서 URL을 제외하고 마지막 이름만 담기
for row in result:
    # 조회된 사용자 URI를 문자열로 변환
    person_uri = str(row[0])

    # '/' 기준 마지막 값인 사용자 이름만 추출
    person_name = person_uri.split('/')[-1]

    # 부산 사용자 이름을 집합에 추가
    busan_users.add(person_name)

# 부산 사용자 이름과 인원 확인
display(busan_users)
display(len(busan_users))
```

### 결과

```text
{'수아', '지아', '지호'}
3
```

## 5. SPARQL 선택 · 존재 확인

### OPTIONAL로 없는 값 처리

관심 브랜드가 없는 사용자도 결과에서 제외하지 않고 조회.

```python
# 쿼리문 작성
query = QUERY_PREFIX + """
SELECT ?person ?brand
WHERE {
    # 모든 사용자 찾기
    ?person a ex:사용자 .

    # 관심 브랜드가 있으면 함께 가져오기
    # 없어도 사용자는 결과에서 제외하지 않음
    OPTIONAL {
        ?person ex:관심 ?brand .
    }
}
"""

# 쿼리문 실행, 결과 담기
result = sns_graph.query(query)

# 사용자 이름 : 관심 브랜드 이름 형태로 담을 사전
interest_map = {}

for row in result:
    # 사용자 URI를 문자열로 변환
    person_uri = str(row[0])

    # URL의 마지막 부분인 사용자 이름만 추출
    person_name = person_uri.split('/')[-1]

    # 관심 브랜드가 없는 경우
    if row[1] is None:
        brand_name = '없음'

    # 관심 브랜드가 있는 경우
    else:
        brand_uri = str(row[1])
        brand_name = brand_uri.split('/')[-1]

    # 사용자 이름을 키, 브랜드 이름을 값으로 저장
    interest_map[person_name] = brand_name


display(interest_map)
display(len(interest_map))
```

### 클래스 계층 따라가기

타입 사이의 상하 관계(`rdfs:subClassOf`)를 따라 상위 클래스 `계정`에 해당하는 노드 조회.

```python
# 타입 계층을 따라가 계정에 해당하는 노드 찾기
query = QUERY_PREFIX + """
SELECT ?x
WHERE {
    # x의 타입을 찾고,
    # 그 타입에서 rdfs:subClassOf를 0홉 이상 따라가
    # 최종적으로 계정에 도달하는지 확인
    ?x a/rdfs:subClassOf* ex:계정 .
}
"""

# 쿼리 실행
result = sns_graph.query(query)

# 계정에 해당하는 이름을 중복 없이 저장
accounts = set()

for row in result:
    # 조회된 URI를 문자열로 변환
    account_uri = str(row[0])

    # URL의 마지막 부분인 이름만 추출
    account_name = account_uri.split('/')[-1]

    # 이름을 집합에 추가
    accounts.add(account_name)

# 결과 확인
display(accounts)
display(len(accounts))
```
