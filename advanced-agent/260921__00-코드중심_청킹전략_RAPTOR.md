# 청킹 전략과 계층 검색 정리

Fixed Chunking부터 Semantic Chunking, Parent-Child, Sentence Window, Auto-Merging, RAPTOR까지의 흐름을 하나로 정리한다.  
LV1의 원리 확인용 코드는 핵심 계산만 남기고, 실제 구현은 LV2 코드를 중심으로 구성한다.  
중복되는 데이터 로딩·문서 변환 코드는 공통 준비 단계에서 한 번만 정의한다.

## 0. 공통 준비

원문을 `Document`로 만들고, 청크가 원래 문서의 어느 위치에서 왔는지 추적할 수 있도록 `source_id`, `start_index`, `end_index`를 metadata에 저장한다.

```python
import json
import os
import sys
from pathlib import Path

from dotenv import find_dotenv, load_dotenv
from langchain_core.documents import Document
from langchain_openai import ChatOpenAI, OpenAIEmbeddings
from llama_index.embeddings.langchain import LangchainEmbedding

material_dir = Path(".")
# 프로젝트 루트를 import 탐색 경로에 추가해 util.py 등을 불러올 수 있게 함
sys.path.insert(0, str(material_dir.resolve()))
data_dir = material_dir / "data"


def read_json(name):
    # JSON 파일을 읽어 Python 객체로 변환
    return json.loads((data_dir / name).read_text(encoding="utf-8"))


def make_documents(records):
    # 원문과 출처·위치 정보를 함께 가진 LangChain Document로 변환
    return [
        Document(
            page_content=record["text"],
            metadata={
                "source_id": record["doc_id"],
                "title": record["title"],
                "url": record["url"],
                "start_index": 0,
                "end_index": len(record["text"]),
            },
        )
        for record in records
    ]


records = read_json("lv2_docs.json")
documents = make_documents(records)
# source_id로 원문을 빠르게 다시 찾기 위한 조회용 딕셔너리
records_by_id = {record["doc_id"]: record for record in records}

# 모든 검색 전략에서 동일하게 사용할 테스트 질문
question_text = "Which early release introduced Fusion indexes by default?"

from util import set_end_offsets, nodes_to_documents, restore_positions
```

### 임베딩·LLM 설정

Semantic Chunking, Vector Search, RAPTOR에서 공통으로 사용한다.

# 현재 작업 디렉터리를 기준으로 .env를 찾아 환경변수 로드

)

index_embedding = LangchainEmbedding(embedding_model)

## 1. Fixed Chunking과 원문 위치 추적

고정 길이로 문서를 나누되 overlap을 두어 경계 문맥을 일부 보존한다. `start_index`와 `end_index`를 함께 기록하면 검색된 청크를 원문 구간과 다시 연결할 수 있다.

```python
from langchain_text_splitters import RecursiveCharacterTextSplitter

fixed_splitter = RecursiveCharacterTextSplitter(
    chunk_size=180,
    # 인접 청크가 30자씩 겹치도록 해 경계 문맥 손실을 줄임
    chunk_overlap=30,
    # 문자 길이 기준으로만 자르기 위해 빈 문자열 separator 사용
    separators=[""],
    strip_whitespace=False,
    # 각 청크의 원문 시작 위치를 metadata에 자동 기록
    add_start_index=True,
)

# start_index와 청크 길이를 이용해 end_index까지 계산
fixed_parts = set_end_offsets(
    fixed_splitter.split_documents(documents)
)

source_matches = [
    doc.page_content
    == records_by_id[doc.metadata["source_id"]]["text"][
        doc.metadata["start_index"] : doc.metadata["end_index"]
    ]
    for doc in fixed_parts
]

release0_starts = [
    doc.metadata["start_index"]
    for doc in fixed_parts
    if doc.metadata["source_id"] == "release0"
]

print("청크 수:", len(fixed_parts))
print("release0 시작 위치:", release0_starts)
print("원문 일치:", all(source_matches))
```

**결과**

```text
청크 수: 15
release0 시작 위치: [0, 150]
원문 일치: True
```

## 2. Semantic Chunking

Semantic Chunking은 이웃 문장 사이 의미 거리가 큰 지점을 경계로 사용한다. 백분위 기준이 낮을수록 더 많은 지점이 경계가 되기 쉽다.

### 백분위 경계 원리

```python
import numpy as np

neighbor_distances = [0.10, 0.20, 0.30, 0.60, 0.80]
# 거리의 50백분위를 의미 경계 기준으로 사용
percentile = 50

distance_threshold = np.percentile(neighbor_distances, percentile)
# 임계값보다 큰 거리의 다음 문장을 새 청크 시작점으로 판단
boundary_indices = [
    index + 1
    for index, distance in enumerate(neighbor_distances)
    if distance > distance_threshold
]

boundary_counts = {
    p: sum(
        distance > np.percentile(neighbor_distances, p)
        for distance in neighbor_distances
    )
    for p in [50, 80]
}

print(distance_threshold)
print(boundary_indices)
print(boundary_counts)
```

**결과**

```text
0.3
[4, 5]
{50: 2, 80: 1}
```

### 실제 Semantic Splitter 적용

```python
from llama_index.core import Document as IndexDocument
from llama_index.core.node_parser import SemanticSplitterNodeParser


def to_index_documents(documents):
    result = [IndexDocument.from_langchain_format(doc) for doc in documents]

    for doc in result:
        # LlamaIndex 내부 문서 ID를 원문의 source_id와 맞춰 출처 추적 유지
        doc.id_ = doc.metadata["source_id"]
        # 제목·URL·위치 metadata가 임베딩/LLM 입력 텍스트에 섞이지 않도록 제외
        doc.excluded_embed_metadata_keys = list(doc.metadata)
        doc.excluded_llm_metadata_keys = list(doc.metadata)

    return result


index_documents = to_index_documents(documents)

percentile50_parser = SemanticSplitterNodeParser(
    # 현재 문장과 인접 문장 1개를 묶어 의미 변화를 비교
    buffer_size=1,
    # 낮은 백분위일수록 경계를 더 많이 생성하는 편
    breakpoint_percentile_threshold=50,
    embed_model=index_embedding,
)

percentile80_parser = SemanticSplitterNodeParser(
    buffer_size=1,
    # 50보다 높은 임계값이므로 일반적으로 더 적게 분할됨
    breakpoint_percentile_threshold=80,
    embed_model=index_embedding,
)

nodes50 = percentile50_parser.get_nodes_from_documents(index_documents)
nodes80 = percentile80_parser.get_nodes_from_documents(index_documents)

percentile50_chunks = restore_positions(
    nodes_to_documents(nodes50),
    records_by_id,
    sequential=True,
)

percentile80_chunks = restore_positions(
    nodes_to_documents(nodes80),
    records_by_id,
    sequential=True,
)

print("50 백분위:", len(percentile50_chunks))
print("80 백분위:", len(percentile80_chunks))
```

**결과 확인**

```text
50 백분위 청크 수 >= 80 백분위 청크 수
80 백분위 청크 수 > 원문 수(5)
```

## 3. Parent-Child Retrieval

작은 child 청크로 유사도 검색을 수행하고, 최종 문맥은 child가 가리키는 원본 parent 문서로 반환한다. 검색 정밀도와 넓은 문맥을 함께 확보할 수 있다.

```python
from langchain_chroma import Chroma
from langchain_core.stores import InMemoryStore
from langchain_classic.retrievers import ParentDocumentRetriever

pc_store = Chroma(
    collection_name="day46_lv2_parent",
    embedding_function=embedding_model,
)
# 이전 실행 데이터가 검색 결과에 섞이지 않도록 컬렉션 초기화
pc_store.reset_collection()

# 실제 parent 원문은 별도 문서 저장소에서 관리
pc_docstore = InMemoryStore()

child_splitter = RecursiveCharacterTextSplitter(
    # 검색 정밀도를 위해 parent보다 작은 child 청크 생성
    chunk_size=120,
    chunk_overlap=20,
)

pc_retriever = ParentDocumentRetriever(
    vectorstore=pc_store,
    docstore=pc_docstore,
    child_splitter=child_splitter,
    # child metadata에 parent를 가리키는 ID를 저장할 key
    id_key="parent_id",
    # 유사도가 높은 child 3개를 먼저 검색
    search_kwargs={"k": 3},
)

pc_retriever.add_documents(
    documents,
    # parent ID를 원문의 source_id와 동일하게 지정
    ids=[doc.metadata["source_id"] for doc in documents],
)

child_hits = pc_store.similarity_search(question_text, k=3)
parent_hits = pc_retriever.invoke(question_text)

print("검색된 child:", len(child_hits))
print("반환된 parent:", len(parent_hits))
print(
    "부모 연결 일치:",
    {doc.metadata["parent_id"] for doc in child_hits}
    == {doc.metadata["source_id"] for doc in parent_hits},
)
```

**결과 확인**

```text
검색된 child: 3
반환된 parent: 1~3
부모 연결 일치: True
```

## 4. Sentence Window Retrieval

검색은 한 문장 단위로 수행하고, 답변에 전달할 때는 검색 문장 앞뒤의 문장을 함께 붙인다. 검색 단위는 작게 유지하면서 최종 문맥은 넓힐 수 있다.

```python
from llama_index.core.node_parser import SentenceWindowNodeParser

# 각 검색 문장 기준 앞뒤 1문장까지 window metadata에 함께 저장
window_parser = SentenceWindowNodeParser.from_defaults(window_size=1)

sentence_nodes = window_parser.get_nodes_from_documents(index_documents)

sentence_documents = restore_positions(
    nodes_to_documents(sentence_nodes),
    records_by_id,
    sequential=True,
)

sentence_store = Chroma(
    collection_name="day46_lv2_sentences",
    embedding_function=embedding_model,
)
sentence_store.reset_collection()
sentence_store.add_documents(sentence_documents)

# 검색은 한 번만 수행
sentence_hits = sentence_store.similarity_search(question_text, k=3)

# 검색된 문장 그대로 사용
window0 = restore_positions(sentence_hits, records_by_id)

# 같은 검색 결과에서 window만 본문으로 확장
expanded_hits = [
    Document(
        # 검색된 한 문장 대신 미리 저장된 앞뒤 문맥(window)을 본문으로 사용
        page_content=hit.metadata["window"],
        metadata=hit.metadata.copy(),
    )
    for hit in sentence_hits
]

window1 = restore_positions(expanded_hits, records_by_id)

chars0 = sum(len(doc.page_content) for doc in window0)
chars1 = sum(len(doc.page_content) for doc in window1)

print("문장 검색 문맥:", chars0)
print("Window 확장 문맥:", chars1)
print("문맥 증가:", chars1 > chars0)
```

**결과 확인**

```text
문맥 증가: True
```

## 5. Auto-Merging Retrieval

문서를 부모-자식 계층으로 나눈 뒤 leaf만 검색한다. 같은 부모의 여러 자식이 충분히 검색되면 개별 자식 대신 부모 문맥으로 병합한다.

### 병합 기준 확인

```python
retrieved_child_ids = ["child_0", "child_0", "child_1"]
total_children = 4
merge_threshold = 0.5

# 같은 child가 여러 번 검색돼도 한 번만 계산
unique_child_ids = set(retrieved_child_ids)
hit_ratio = len(unique_child_ids) / total_children
# 기준과 같을 때가 아니라 초과할 때만 parent로 병합
should_merge = hit_ratio > merge_threshold

print(hit_ratio)
print(should_merge)
```

**결과**

```text
0.5
False
```

기준과 같은 값은 병합하지 않고, 기준을 초과할 때만 병합한다.

### 계층 검색 구현

```python
import chromadb
from llama_index.core import StorageContext, VectorStoreIndex
from llama_index.core.node_parser import HierarchicalNodeParser, get_leaf_nodes
from llama_index.core.retrievers import AutoMergingRetriever
from llama_index.vector_stores.chroma import ChromaVectorStore

auto_parser = HierarchicalNodeParser.from_defaults(
    # 256자 parent와 64자 child의 2단계 계층 생성
    chunk_sizes=[256, 64],
    chunk_overlap=0,
    include_prev_next_rel=False,
)

auto_nodes = auto_parser.get_nodes_from_documents(index_documents)
auto_leaf_nodes = get_leaf_nodes(auto_nodes)

# 실습용 메모리 기반 Chroma 클라이언트: 프로세스 종료 시 데이터 소멸
client = chromadb.EphemeralClient()

try:
    client.delete_collection("day46_lv2_auto")
except Exception:
    pass

auto_collection = client.create_collection(
    name="day46_lv2_auto",
    embedding_function=None,
    configuration={"hnsw": {"space": "cosine"}},
)

auto_vector_store = ChromaVectorStore(
    chroma_collection=auto_collection,
)

auto_storage = StorageContext.from_defaults(
    vector_store=auto_vector_store,
)

# 부모까지 포함한 전체 계층은 docstore에 보관
auto_storage.docstore.add_documents(auto_nodes)

# 검색 후보는 leaf만 벡터 인덱스에 적재해 세밀한 유사도 검색 수행
auto_index = VectorStoreIndex(
    auto_leaf_nodes,
    storage_context=auto_storage,
    embed_model=index_embedding,
)

base_retriever = auto_index.as_retriever(similarity_top_k=3)

auto_retriever = AutoMergingRetriever(
    base_retriever,
    auto_storage,
    # 같은 parent의 leaf가 절반을 초과해 검색되면 상위 문맥으로 병합
    simple_ratio_thresh=0.5,
)

auto_hits = auto_retriever.retrieve(question_text)

auto_contexts = restore_positions(
    nodes_to_documents(
        [hit.node for hit in auto_hits],
        auto_nodes,
        records_by_id,
    ),
    records_by_id,
)

print("검색 결과 수:", len(auto_hits))
print("Chroma에는 leaf만 저장:", len(auto_collection.get()["ids"]) == len(auto_leaf_nodes))
```

**결과 확인**

```text
Chroma에는 leaf만 저장: True
부모를 포함한 전체 노드는 docstore에 저장됨
검색 결과는 원문의 start_index/end_index와 다시 연결됨
```

## 6. RAPTOR: 문장 → 군집 → 요약 계층

RAPTOR는 원문 leaf를 임베딩해 비슷한 내용끼리 묶고, 각 군집을 요약해 상위 계층을 만든다. 검색된 요약은 `leaf_ids`를 통해 최종 원문 근거까지 다시 내려갈 수 있어야 한다.

```python
from sklearn.cluster import KMeans
from sklearn.preprocessing import normalize
from langchain_core.prompts import ChatPromptTemplate
from langchain_core.output_parsers import StrOutputParser

leaves = [
    {
        "node_id": f"leaf_{index}",
        "level": 0,
        "text": doc.page_content,
        "child_ids": [],
        # 상위 요약에서도 최종 원문까지 역추적할 수 있도록 leaf ID 보존
        "leaf_ids": [f"leaf_{index}"],
        "source": doc.metadata.copy(),
    }
    for index, doc in enumerate(sentence_documents)
]

leaf_vectors = embedding_model.embed_documents(
    [leaf["text"] for leaf in leaves]
)

labels = KMeans(
    # 실습에서는 leaf를 3개 의미 군집으로 묶음
    n_clusters=3,
    # 실행마다 같은 군집 결과를 재현하기 위한 시드 고정
    random_state=0,
    n_init=10,
).fit_predict(normalize(leaf_vectors))

groups = {
    label: [
        leaves[index]
        for index, current_label in enumerate(labels)
        if current_label == label
    ]
    for label in sorted(set(labels))
}

summary_prompt = ChatPromptTemplate.from_messages([
    (
        "system",
        "주어진 문맥의 핵심 사실을 짧게 요약한다. 서로 다른 버전의 내용을 임의로 합치지 않는다.",
    ),
    ("human", "{context}"),
])

# 프롬프트 → LLM → 문자열 출력 파서 순서로 요약 체인 구성
summary_chain = summary_prompt | llm | StrOutputParser()


def summarize_groups(groups, level, summary_chain):
    summaries = []

    for group_id, members in sorted(groups.items()):
        sections = []

        for node in members:
            title = node["source"]["title"] if node["level"] == 0 else "하위 요약"
            sections.append(
                f"[{node['node_id']}] {title}\n{node['text']}"
            )

        summary = summary_chain.invoke({
            "context": "\n\n".join(sections)
        })

        summaries.append({
            "node_id": f"summary_{level}_{group_id}",
            "level": level,
            "text": summary,
            # 바로 아래 계층과의 연결 관계 저장
            "child_ids": [node["node_id"] for node in members],
            # 하위 노드가 가진 leaf ID를 모두 모아 최종 원문 연결 유지
            "leaf_ids": [
                leaf_id
                for node in members
                for leaf_id in node["leaf_ids"]
            ],
        })

    return summaries


level1 = summarize_groups(groups, 1, summary_chain)
leaves_by_id = {leaf["node_id"]: leaf for leaf in leaves}

print("군집 수:", len(groups))
print("L1 요약 수:", len(level1))
```

**결과 확인**

```text
군집 수: 3
L1 요약 수: 3
각 leaf는 정확히 하나의 L1 요약에 연결됨
```

상위 L2 계층을 만들 때는 L1의 `text`를 다시 임베딩하고 군집화한다. `child_ids`는 바로 아래 요약 계층을 연결할 때 사용하고, `leaf_ids`는 최종 원문 근거를 찾을 때 사용한다.

## 7. 요약 결과에서 원문 근거 복원

검색 결과에 summary와 leaf가 섞여 있더라도 최종 근거는 원문 leaf ID로 통일한다. 이미 추가한 leaf는 다시 넣지 않아 중복 근거를 방지한다.

```python
summary_nodes = {
    "summary_1_0": {
        "level": 1,
        "leaf_ids": ["leaf_0", "leaf_2"],
    },
    "summary_1_1": {
        "level": 1,
        "leaf_ids": ["leaf_1", "leaf_3", "leaf_4"],
    },
    "summary_2_0": {
        "level": 2,
        "leaf_ids": ["leaf_0", "leaf_1", "leaf_2", "leaf_3", "leaf_4"],
    },
}

retrieved_ids = ["summary_1_1", "leaf_3", "leaf_2"]

evidence_ids = []

for retrieved_id in retrieved_ids:
    # summary면 연결된 원문 leaf들로 펼치고, leaf면 자기 자신을 사용
    ids = (
        summary_nodes[retrieved_id]["leaf_ids"]
        if retrieved_id in summary_nodes
        else [retrieved_id]
    )

    for leaf_id in ids:
        # 같은 leaf가 여러 경로에서 나와도 최종 근거에는 한 번만 추가
        if leaf_id not in evidence_ids:
            evidence_ids.append(leaf_id)

print(evidence_ids)
```

**결과**

```text
['leaf_1', 'leaf_3', 'leaf_4', 'leaf_2']
```

`child_ids`는 계층 구조를 따라 다음 요약을 만들 때 사용한다. `leaf_ids`는 검색된 요약에서 실제 원문 근거를 복원할 때 사용한다. 출처 연결이 유지되더라도 요약 과정에서 원문의 조건·수치가 생략될 수 있으므로, 최종 답변 근거는 leaf 원문으로 다시 확인하는 편이 안전하다.
