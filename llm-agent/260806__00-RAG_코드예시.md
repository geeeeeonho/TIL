# 1. 기본 설정

```python
# [제공 코드] 이 단원에 필요한 라이브러리와 한국어 임베딩 모델을 준비합니다.
import numpy as np
import pandas as pd
from sentence_transformers import SentenceTransformer
from sklearn.metrics.pairwise import cosine_similarity
import chromadb


# 지난 단원에서 배운 한국어 임베딩 모델
# 문장 한 개를 768차원 벡터로 바꿉니다.
# 처음 부를 때 모델을 내려받느라 조금 걸릴 수 있어요.
emb_model = SentenceTransformer(
    'jhgan/ko-sroberta-multitask'
)

print(
    '임베딩 모델 준비 완료 — 벡터 차원:',
    emb_model.get_embedding_dimension()
)


# [제공 코드] 도서 데이터를 살펴봅니다.
books = pd.read_csv('data/books.csv')

print("행·열 크기:", books.shape)

print("\n[장르별 개수]")
print(books["genre"].value_counts())

print("\n[앞 5행] head()")
display(books.head())
```

# 3. 코사인 유사도와 임베딩

## 3.1 질문과 가장 비슷한 책 1권 찾기

```python
# 검색에 사용할 질문 문장
words1 = "무서운 살인 사건을 파헤치는 이야기"


# 질문 문장 하나를 임베딩하여 질문 벡터 생성
# normalize_embeddings=True:
# 벡터의 길이를 1로 정규화
words1_vec = emb_model.encode(
    words1,
    normalize_embeddings=True
)


# 질문 벡터를 책의 임베딩 벡터 목록과 비교
#
# 1. cosine_similarity()는 2차원 입력을 요구함
#    [words1_vec]으로 감싸
#    (768,)을 (1, 768) 형태로 변환
#
# 2. 질문 벡터와 모든 책의 임베딩 사이
#    코사인 유사도를 계산
#
# 3. 결과의 [0]을 선택하여
#    2차원 결과를 1차원 유사도 배열로 변환
sims = cosine_similarity(
    [words1_vec],
    book_emb
)[0]


# 유사도 배열에서 가장 큰 값이 있는 위치 번호
best_idx = np.argmax(sims)


# 가장 유사한 책의 장르
best_genre = books.loc[
    best_idx,
    'genre'
]

print(best_idx)
print(best_genre)
```

### 실행 결과

```text
16
추리
```

## 3.2 특정 책과 가장 비슷한 책 3권 찾기

추천 시스템의 기본 형태.

```python
# 3번 책의 임베딩과
# 전체 책의 임베딩 간 코사인 유사도 계산
#
# book_emb[3]은 (768,) 형태의 1차원 벡터이므로
# [book_emb[3]]으로 감싸
# (1, 768) 형태의 2차원 입력으로 변환
#
# 결과의 [0]을 사용해
# 1차원 유사도 배열만 꺼냄
sims3 = cosine_similarity(
    [book_emb[3]],
    book_emb
)[0]


# 유사도에 음수를 붙여 큰 값부터 정렬하고
# 정렬된 위치 번호를 반환
sims3_sort = np.argsort(-sims3)


# 비교 대상인 자기 자신인 3번 위치를 제외
sims3_sort = sims3_sort[
    sims3_sort != 3
]


# 남은 결과 중 유사도가 가장 높은
# 책 3개의 위치 번호 선택
top3 = sims3_sort[:3]

print(top3)
```

### 실행 결과

```text
[ 8 14 13]
```

# 4. ChromaDB 컬렉션 만들기

## 4.1 저장 구조

```text
client: EphemeralClient
│
└── book_lib: Collection
    │
    ├── 컬렉션 이름
    │   └── book_lib
    │
    ├── 컬렉션 설정
    │   └── hnsw:space = cosine
    │
    ├── HNSW 벡터 인덱스
    │   └── 768차원 임베딩을 이용해 유사한 책 검색
    │
    └── 도서 레코드 17개
        │
        ├── id
        ├── embedding: 숫자 768개
        ├── document: 책 설명
        └── metadata
            ├── title
            └── genre
```

---

## 4.2 클라이언트·컬렉션 생성과 문서 적재

```python
# 메모리에만 존재하는 ChromaDB 벡터 DB 클라이언트 생성
# 커널이나 프로그램이 종료되면 저장된 데이터도 사라짐
client = chromadb.EphemeralClient()


# 코사인 거리를 사용하는 book_lib 컬렉션 생성
#
# 아래 metadata는 책 한 권의 정보가 아니라
# 컬렉션 전체의 벡터 검색 설정
#
# hnsw:
# 벡터를 빠르게 검색하기 위한 인덱스 방식
#
# space:
# 벡터 사이의 거리를 어떤 방법으로 계산할지 지정
#
# cosine:
# 벡터 사이의 코사인 거리를 사용
# 코사인 거리 = 1 - 코사인 유사도
book_lib = client.get_or_create_collection(
    'book_lib',
    metadata={
        'hnsw:space': 'cosine'
    }
)


# 각 책의 제목과 장르를 딕셔너리 하나로 묶어
# 책의 개수만큼 메타데이터 딕셔너리 목록 생성
book_metadata = [
    {
        'title': title,  # 제목 열 설정
        'genre': genre,  # 장르 열 설정
    }
    for title, genre in zip(
        books['title'],
        books['genre']
    )
]


# 도서 ID, 임베딩 벡터, 설명, 메타데이터를
# 같은 위치끼리 하나의 도서 레코드로 연결하여 저장
#
# ids[i], embeddings[i], documents[i], metadatas[i]
# → 같은 위치의 값이 한 권의 책으로 연결
book_lib.add(
    ids=books['id'].tolist(),                  # 도서별 고유 식별자 목록
    embeddings=book_emb,                       # 도서 17권의 임베딩 벡터
    documents=books['description'].tolist(),  # 도서 설명 문장 목록
    metadatas=book_metadata,                   # 도서별 제목·장르 정보
)
```

# 5. 컬렉션에 질문하기

컬렉션 자체가 거리 계산과 검색을 수행함.  
직접 코사인 유사도를 계산하지 않고 `query()`를 사용할 수 있음.

## 5.1 동물이 나오는 이야기책 검색

```python
# 검색할 문장
word3 = "아이에게 읽어 줄 동물이 나오는 이야기책"


# 문자열 임베딩
# 문자열을 리스트로 넣었으므로
# 결과는 (1, 768) 형태의 2차원 배열
q_emb = emb_model.encode(
    [word3],
    normalize_embeddings=True
)

print(
    'q_emb.shape:',
    q_emb.shape
)


# 컬렉션 검색 사용
# 가장 가까운 책 3개 꺼내기
res = book_lib.query(
    query_embeddings=q_emb,
    n_results=3
)


# query()는 여러 질문을 처리할 수 있어
# 결과를 질문별 이중 리스트로 반환
#
# [0]으로 첫 번째 질문의 검색 결과를 꺼냄
hit_ids = res['ids'][0]


# 첫 번째 질문으로 검색된 책의 제목·장르 정보
hit_metadatas = res['metadatas'][0]


# 첫 번째 질문과 검색된 책 사이의 코사인 거리
# 거리가 작을수록 질문과 더 가까운 책
hit_distances = res['distances'][0]


# 검색된 책의 ID, 제목, 장르, 거리 출력
for book_id, metadata, distance in zip(
    hit_ids,
    hit_metadatas,
    hit_distances
):
    print(
        'ID:', book_id,
        '| 제목:', metadata['title'],
        '| 장르:', metadata['genre'],
        '| 거리:', distance
    )
```

### 실행 결과

```text
q_emb.shape: (1, 768)
ID: b12 | 제목: 암호를 푸는 소녀 | 장르: 추리 | 거리: 0.6014592051506042
ID: b08 | 제목: 숲속 동물 재판 | 장르: 동화 | 거리: 0.6039597988128662
ID: b13 | 제목: 무지개 마을 우체부 | 장르: 동화 | 거리: 0.612086296081543
```

    in res['metadatas'][0]

faqs_question = faqs[

## 9.2 검색: 컬렉션 쿼리 함수

```python
def search_faq(question):
    # 받은 문장 임베딩 진행
    q_emb = emb_model.encode(
        [question],
        normalize_embeddings=True
    )

    # faq_bot 컬렉션에서
    # 가장 가까운 FAQ 1건 검색
    res = faq_bot.query(
        query_embeddings=q_emb,  # 해당 문장
        n_results=1,             # 가장 가까운 값 1개
    )

    # 첫 번째 질문의 첫 번째 검색 결과에서
    # category와 answer가 들어 있는
    # 메타데이터 딕셔너리 추출
    response = res[
        'metadatas'
    ][0][0]

    return response
```

    )

## 10.2 선택 하나를 처리하는 함수

```python
# 선택 번호 처리 함수
def handle_choice(n):
    # 1부터 질문 개수까지의 번호인지 확인
    if n in range(
        1,
        len(PRESET_QUESTIONS) + 1
    ):
        # 번호에 해당하는 질문 문장 선택
        # 번호 1은 리스트 인덱스 0이므로 n - 1 사용
        #
        # 수정:
        # 원본은 faqs['question'][n - 1]을 사용해
        # 화면의 PRESET_QUESTIONS와 실제 검색 질문이 달라질 수 있었음
        question = PRESET_QUESTIONS[
            n - 1
        ]

        # 질문 문장으로 가장 가까운 FAQ 검색
        # search_faq() 함수 호출
        #
        # 결과는 category와 answer가 담긴 딕셔너리
        result = search_faq(question)

        # 분류와 답변을 하나의 문자열로 만들어 반환
        return (
            f"[{result['category']}] "
            f"{result['answer']}"
        )

    return '없는 번호입니다.'
```

테스트:

```python
print(handle_choice(2))
print(handle_choice(100))
```

없는 번호입니다.

> 원본에서는 `handle_choice(2)`가 실제 FAQ 데이터의 두 번째 질문을 검색해 배송 답변을 반환했음.  
> 현재 코드는 화면에 표시한 두 번째 메뉴인 `반품하면 환불금은 언제 돌려받나요`를 검색하도록 수정함.

## 10.3 메뉴 루프 입력

```python
search_count = 0  # 검색 횟수


while True:
    # 질문 목록을 1번부터 번호를 붙여 출력
    for number, question in enumerate(
        PRESET_QUESTIONS,
        start=1
    ):
        print(
            f'{number}. {question}'
        )

    # 종료 메뉴 출력
    print('0. 종료')

    # 번호 입력받기
    try:
        i = int(
            input('번호 선택: ')
        )
    except ValueError:
        print('숫자를 입력하세요.')
        continue

    # 0을 입력하면 종료
    if i == 0:
        print(
            '검색봇을 종료합니다. '
            f'(총 {search_count}번 검색)'
        )
        break

    # 선택한 번호의 처리 결과를 변수에 저장
    result = handle_choice(i)

    # 처리 결과 출력
    print(result)

    # 정상적인 번호를 선택했을 때만
    # 검색 횟수 증가
    if result != '없는 번호입니다.':
        search_count += 1
```
