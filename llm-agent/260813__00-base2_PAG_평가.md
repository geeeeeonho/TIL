# 검색 평가 지표

## Hit@K 평가지표

상위 K개 중 정답이 하나라도 있는지 확인.

```python
# 순수 검색 평가 지표 함수: Hit@K
# 상위 K개 안에 정답이 하나라도 있는지 확인
def hit_at_k(ranked, gold, k) -> float:
    for i in ranked[:k]:
        if i in gold:
            return 1.0  # 정답이 하나라도 있으면 바로 1 반환

    return 0.0  # 정답이 하나도 없으면 0 반환

print(hit_at_k(['a', 'b', 'c'], ['b', 'x', 'y'], 3))
print(hit_at_k(['a', 'b', 'c'], ['b', 'x', 'y'], 1))
```

**결과**

```text
1.0
0.0
```

## Precision@K 평가지표

상위 K개 중 정답이 몇 개인지 비율로 확인.

```python
# 함수 제작
def precision_at_k(ranked, gold, k) -> float:
    count = 0

    # 정답과 검색 결과를 하나씩 비교
    for i in ranked[:k]:
        if i in gold:
            count += 1  # 정답이면 1개 추가

    return count / k  # 상위 K개 중 정답의 비율 반환

# 테스트
print(precision_at_k(['a', 'b', 'c'], ['b', 'x', 'y'], 3))
print(precision_at_k(['a', 'b', 'c'], ['b', 'x', 'y'], 1))
```

**결과**

```text
0.3333333333333333
0.0
```

## Recall@K 평가지표

찾았어야 할 전체 정답 중 몇 개를 검색했는지 확인.

```python
# 함수 제작
def recall_at_k(ranked, gold, k) -> float:
    count = 0

    # 정답과 검색 결과를 하나씩 비교
    for i in ranked[:k]:  # 검색 결과 중 상위 K개만 사용
        if i in gold:
            count += 1  # 정답이면 1개 추가

    return count / len(gold)  # 전체 정답 중 검색된 정답의 비율 반환

# 테스트
print(recall_at_k(['a', 'b', 'c'], ['b', 'x', 'y'], 3))
print(recall_at_k(['a', 'b', 'c'], ['b'], 3))
```

**결과**

```text
0.3333333333333333
1.0
```

## MRR@K 평가지표

정답이 얼마나 높은 순위에 있었는지 확인.

- 1위: `1.0`
- 2위: `0.5`
- 4위: `0.25`

```python
def mrr_at_k(ranked, gold, k) -> float:
    # 현재 몇 위인지 저장
    count = 0

    # 검색 결과 중 상위 K개만 확인
    for i in ranked[:k]:
        # 한 칸 내려갈 때마다 순위 +1
        count += 1

        # 처음 만난 정답이라면 해당 순위의 역수 반환
        if i in gold:
            return 1 / count

    # 상위 K개 안에서 정답을 못 찾았다면
    return 0.0

# 테스트
print(mrr_at_k(['a', 'b', 'c'], ['b', 'x', 'y'], 3))
print(mrr_at_k(['b', 'a', 'c'], ['b', 'x'], 3))
```

**결과**

```text
0.5
1.0
```

```python
# 평가할 검색 결과 개수
# 각 질문마다 검색 결과 상위 3개까지만 보고 평가
K = 3

# 문항별 평가 결과를 하나씩 저장할 빈 리스트
# 나중에 이 리스트를 DataFrame으로 바꿀 예정
results = []

# 평가셋(evalset)의 질문을 한 행씩 반복
# evalset은 "시험 문제 + 정답표" 역할
for _, row in evalset.iterrows():

    # 1. 평가 질문을 검색기에 넣어서 상위 3개 조각 id를 가져옴
    # 예: query = "쇼핑몰 계정 아이디와 결제 기록도 개인정보인가요?"
    #
    # 검색 결과 예: ranked = ['q1-0', 'q5-0', 'q64-0']
    #
    # 중요한 점: 한 질문당 search_ids()는 딱 한 번만 실행
    #
    # 이 ranked 결과 하나를 네 평가 지표가 모두 같이 사용
    ranked = search_ids(row['query'], K)

    # 2. 정답 조각 id를 리스트로 변환
    #
    # evalset의 gold_chunks는 문자열 형태
    # 예: 'q1-0|q64-0'
    #
    # split('|')을 사용하면 ['q1-0', 'q64-0'] 형태의 리스트로 변환됨
    #
    # 이 리스트가 실제 채점할 때 사용하는 정답표
    gold = row['gold_chunks'].split('|')

    # 3. 현재 질문에 대한 네 가지 검색 평가 지표 계산
    #
    # ranked : 검색기가 찾은 상위 K개(3개) 조각
    # gold   : 실제 정답 조각 목록
    result = {
        'query_id': row['query_id'],
        'Hit': hit_at_k(ranked, gold, K),      # Hit: 상위 3개 안에 정답이 하나라도 있는가?
        'P': precision_at_k(ranked, gold, K),  # P: 상위 3개 중 정답이 몇 개나 있는가?
        'R': recall_at_k(ranked, gold, K),     # R: 전체 정답 중 몇 개를 찾아냈는가?
        'MRR': mrr_at_k(ranked, gold, K)       # MRR: 처음 나온 정답이 몇 위에 있는가?
    }

    # 4. 현재 질문의 평가 결과를 results 리스트에 저장
    #
    # 반복이 진행되면 results 안에는 질문 25개의 평가 결과가 차례대로 쌓임
    results.append(result)

# 5. 질문별 평가 결과를 DataFrame으로 변환
# 결과 형태 예:
# query_id   Hit     P       R       MRR
# qna01      1.0    0.67    1.0     1.0
# qna02      1.0    0.33    1.0     0.5
# ...
# 한 행이 평가 질문 하나를 의미
eval_results = pd.DataFrame(results)

# 6. 전체 평가 질문에 대한 네 지표의 평균 계산
# 질문별 점수를 모두 구했으므로, 검색기의 전체 성적을 평균으로 계산
eval_summary = {
    'Hit': eval_results['Hit'].mean(),
    'P': eval_results['P'].mean(),
    'R': eval_results['R'].mean(),
    'MRR': eval_results['MRR'].mean()
}

# 7. 검색에 실패한 질문 찾기
# Hit가 0.0이라는 것은 상위 3개 검색 결과 안에 정답 조각이 하나도 없었다는 뜻
# 그 행들만 골라서 query_id만 가져온 뒤, 파이썬 리스트로 변환
missed_qna = eval_results[
    eval_results['Hit'] == 0.0
]['query_id'].tolist()

# 8. 결과 확인
display(eval_results)  # 질문별 상세 평가 결과
print(eval_summary)    # 검색기의 전체 평균 성적
print(missed_qna)      # 상위 3개 안에 정답을 하나도 찾지 못한 질문
```

# K값을 바꿔가며 비교하기

```python
# 평가할 K 값
K_list = [1, 3, 5, 10]

# 질문마다 상위 10개를 한 번만 검색하고 정답과 함께 저장
pairs = []

for _, row in evalset.iterrows():

    # 최대 K인 10개까지 한 번만 검색
    ranked = search_ids(row['query'], 10)

    # 정답 문자열을 리스트로 변환
    gold = row['gold_chunks'].split('|')

    pairs.append((ranked, gold))

# K별 평균 결과 저장
rows = []

for K in K_list:

    # 지표별 점수 저장
    hit_scores = []
    p_scores = []
    r_scores = []
    mrr_scores = []

    # 같은 검색 결과로 K만 바꿔가며 평가
    for ranked, gold in pairs:
        hit_scores.append(hit_at_k(ranked, gold, K))
        p_scores.append(precision_at_k(ranked, gold, K))
        r_scores.append(recall_at_k(ranked, gold, K))
        mrr_scores.append(mrr_at_k(ranked, gold, K))

    # 현재 K의 평균값 저장
    rows.append({
        'K': K,
        'Hit': sum(hit_scores) / len(hit_scores),
        'P': sum(p_scores) / len(p_scores),
        'R': sum(r_scores) / len(r_scores),
        'MRR': sum(mrr_scores) / len(mrr_scores)
    })

# K별 검색 평가표
k_table = pd.DataFrame(rows)
```

## 3단계: 못 찾은 문항 읽어보기

```python
# 검색에 실패한 문항의 query_id를 저장할 리스트
missed = []

for _, row in bs_eval.iterrows():
```
