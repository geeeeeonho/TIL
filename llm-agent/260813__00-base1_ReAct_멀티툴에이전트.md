```python
import requests
from langchain_core.tools import tool

# 도구 제작
@tool
def holiday_names(year: int) -> str:
    """입력한 연도의 한국 공휴일 이름을 조회해 쉼표로 이어 반환합니다."""
    try:
        # API GET 요청
        # API가 공휴일 정보를 "딕셔너리들의 리스트" 형태로 반환
        response = requests.get(
            f'https://date.nager.at/api/v3/PublicHolidays/{year}/KR',
            timeout=10
        )

        # HTTP 오류가 있으면 예외 발생
        response.raise_for_status()

        # 응답을 파이썬 리스트로 변환
        data = response.json()

        # 각 공휴일 딕셔너리에서 localName만 꺼내서 저장
        names = [
            item['localName']
            for item in data
        ]

        # 공휴일 이름들을 쉼표로 연결해 반환
        return ', '.join(names)

    # requests 관련 오류가 발생하면 문자열로 반환
    except requests.RequestException as e:
        return f'공휴일 정보를 불러오지 못했습니다: {e}'


# 테스트
holiday_list = holiday_names.invoke({'year': 2026})
print(holiday_list)
```

## 담당 창구를 돌려주는 도구 + 에이전트

```python
# 담당 창구표
_WINDOW = {
    '여권': '3번 창구',
    '전입신고': '1번 창구',
    '등본 발급': '무인발급기'
}

# 도구 제작
@tool
def find_window(minwon: str) -> str:
    """민원 업무를 처리할 담당 창구를 찾을 때 사용하는 도구입니다.
    사용 가능한 민원 이름은 여권, 전입신고, 등본 발급입니다."""

    # 입력된 민원 이름의 앞뒤 공백 제거
    minwon = minwon.strip()

    # 민원 이름으로 담당 창구 찾기
    window = _WINDOW.get(minwon)

    # 담당 창구를 찾았다면 그대로 반환
    if window is not None:
        return window

    # 표에 없는 민원이라면 선택 가능한 민원 이름 안내
    else:
        return '찾을 수 없는 민원입니다. 여권, 전입신고, 등본 발급 중에서 선택해주세요.'

    model,

    system_prompt='민원 담당 창구를 묻는 질문에는 필요한 도구를 사용하세요.'
)
```

## 문서 경계를 넘지 않게 앞뒤 조각 번호 고르기

모델에게 조각을 넘길 때 문맥 이해를 위해 사용.

```python
# 조각 처리 함수
# chunk_no    : 가져올 중심 청크 번호
# chunk_total : 현재 문서의 전체 청크 개수
# window      : 중심 청크의 앞뒤로 추가로 가져올 청크 개수
def neighbor_numbers(chunk_no, chunk_total, window=1):
    n_list = []

    # chunk_no를 중심으로 앞뒤 window만큼의 번호를 확인
    for i in range(chunk_no - window, chunk_no + window + 1):

        # 0 이상이고 마지막 청크 번호(chunk_total - 1) 이하인 번호만 저장
        if i >= 0 and i < chunk_total:
            n_list.append(i)

    return n_list


# 테스트
print(neighbor_numbers(2, 5))
print(neighbor_numbers(4, 5))
```
