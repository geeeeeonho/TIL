# OpenAI API 코드 예시

## 기본 설정

```python
# [제공 코드] OpenAI 클라이언트 준비 — 이 셀을 먼저 실행하세요.
# .env 파일에 OPENAI_API_KEY를 넣어 두면 아래 코드가 키를 읽어 연결합니다.
# 참고: https://developers.openai.com/api/docs/guides/text

import os
import pandas as pd

from dotenv import load_dotenv
from openai import OpenAI

load_dotenv('.env')       # 같은 폴더의 .env
load_dotenv('../.env')    # 한 단계 상위 폴더의 .env

# max_retries: 분당 토큰 한도(TPM)에 걸리면(429) 잠시 뒤 자동으로 다시 시도한다.
# 이미지는 한 장에 많은 토큰을 사용하므로 여러 장을 연달아 보내면 제한에 걸리기 쉽다.
client = OpenAI(max_retries=8)  # OPENAI_API_KEY를 자동으로 찾아 사용

print(
    '연결 준비 완료 —',
    '키 확인됨' if os.getenv('OPENAI_API_KEY')
    else '키가 없습니다(.env를 확인하세요)'
)
```

## 구조화된 스키마

답변을 정해진 JSON 스키마에 맞춰 받을 수 있다.

```python
# 감정 분석 결과를 담을 JSON 스키마
senti_schema = {
    'type': 'json_schema',
    'json_schema': {
        'name': 'review_sentiment',
        'schema': {
            'type': 'object',
            'properties': {
                'sentiment': {
                    'type': 'string',
                    'enum': ['긍정', '부정', '중립']
                },
                'summary': {'type': 'string'}
            },
            'required': ['sentiment', 'summary'],
            'additionalProperties': False
        },
        'strict': True
    }
}
```

### 사용해 보기

```python
import json

to_messages = [{
    'role': 'user',
    'content': reviews.loc[0, 'content']  # 첫 번째 리뷰 본문 사용
}]

resp = client.chat.completions.create(
    model='gpt-4o-mini',
    temperature=0,
    messages=to_messages,
    response_format=senti_schema
)

# 첫 번째 응답에서 JSON 문자열 추출
answer_json = resp.choices[0].message.content
print(answer_json)

# JSON 문자열을 파이썬 딕셔너리로 변환

# result에는 sentiment와 summary 키가 있음
print(result['sentiment'])
print(result['summary'])
```

## 구조화된 출력: Pydantic 스키마 직접 제작

```python
from pydantic import BaseModel
from typing import Literal

# 구조화된 응답의 형태를 직접 선언
class Sentiment(BaseModel):
    sentiment: Literal['긍정', '부정', '중립']
    reason: str


to_messages = [{
    'role': 'user',
    'content': reviews.loc[1, 'content']  # 두 번째 리뷰 본문 사용
}]

resp = client.chat.completions.parse(
    model='gpt-4o-mini',
    messages=to_messages,
    response_format=Sentiment
)

# 응답에서 Sentiment 타입 객체 추출
parsed = resp.choices[0].message.parsed

print(parsed.sentiment)
print(parsed.reason)
```

```text
부정
교정기의 효과가 없음과 불만족스러운 품질에 대한 실망감을 표현하고 있어, 별 하나도 아깝다는 평가가 이를 반영하고 있습니다.
```

## 파라미터 되짚기: `top_p`

`top_p`를 사용해 응답 생성에 사용할 후보 범위를 조절한다.

```python
to_messages = [{
    'role': 'user',
    'content': '가을에 어울리는 이름을 하나만 지어줘. 이름만 답해.'
}]

# top_p가 좁을 때와 넓을 때의 결과를 저장할 리스트
narrow = []
wide = []
nw = [narrow, wide]

for i in range(len(nw)):
    if i == 0:
        top_n = 0.1  # narrow
    else:
        top_n = 1.0  # wide

    for _ in range(3):
        resp = client.chat.completions.create(
            model='gpt-4o-mini',
            top_p=top_n,
            messages=to_messages
        )

        answer_i = resp.choices[0].message.content
        nw[i].append(answer_i)

print('narrow:', narrow)
print('wide:', wide)
```

## 스트리밍: 답을 실시간으로 받기

`stream=True`를 설정하면 답을 조각 단위로 받을 수 있다.

```python
to_messages = [{
    'role': 'user',
    'content': '커피의 역사를 세 문장으로 알려줘.'
}]

chunks = []

streams = client.chat.completions.create(
    model='gpt-4o-mini',
    stream=True,
    messages=to_messages
)

for chunk in streams:
    content = chunk.choices[0].delta.content

    # 텍스트가 들어 있는 조각만 리스트에 추가
    if content is not None:
        chunks.append(content)

# 문자열 조각들을 하나로 연결
stream_text = ''.join(chunks)

print(chunks)
print(stream_text)
```

```text
['', '커', '피', '는', ' ', '15', '세', '기', ' 아', '라', '비', '아', ' 반', '도', '에서', ' 처음', ' 재', '배', '되고', ' 소비', '되', '기', ' 시작', '했습니다', '.', ...]

커피는 15세기 아라비아 반도에서 처음 재배되고 소비되기 시작했습니다. 이후 17세기 유럽에 전파되어 카페 문화가 형성되었고, 커피는 사회적 교류의 중요한 매개체가 되었습니다. 19세기에는 대량 생산이 가능해지면서 전 세계적으로 보급되었고, 현재는 다양한 종류와 스타일로 즐겨지고 있습니다.
```

## 이미지 분석: 로컬 파일

로컬 파일은 파일 경로만으로 전달할 수 없으므로 이미지를 Base64 문자열로 변환해 데이터 URL을 만든다.

```python
import base64

# 이미지 파일을 바이너리로 열고 데이터 읽기
with open('data/images/food.jpg', 'rb') as image_file:
    image_bytes = image_file.read()

# 이미지 데이터를 Base64 문자열로 변환
base64_image = base64.b64encode(image_bytes).decode('utf-8')

# JPEG 데이터 URL 생성
data_url = f'data:image/jpeg;base64,{base64_image}'

# 텍스트와 이미지를 함께 전달

to_messages = [{
    'role': 'user',
    'content': [
        {
            'type': 'text',
            'text': '이 사진에 무엇이 있나요? 한 문장으로 답해주세요.'
        },
        {
            'type': 'image_url',
            'image_url': {
                'url': data_url
            }
        }
    ]
}]

resp = client.chat.completions.create(
    model='gpt-4o-mini',
    messages=to_messages
)

answer_img = resp.choices[0].message.content
print(answer_img)
```

```text
다양한 신선한 채소와 조미료가 배열된 이미지입니다.
```

# 도구 라우팅과 호출

```python
import json
import random
import pandas as pd


# =========================================================
# 1단계: 도구로 사용할 실제 파이썬 함수 준비
# =========================================================

# 리뷰 데이터 파일을 읽어 DataFrame으로 저장
# count_reviews_by_rating() 함수가 이 데이터를 사용함
reviews = pd.read_csv('data/reviews.csv')


def count_reviews_by_rating(rating):
    """
    전달받은 별점과 같은 리뷰의 개수를 반환한다.

    예:
    rating=3
    → rating 열의 값이 3인 행의 개수를 반환
    """

    # reviews['rating'] == rating
    # → 각 리뷰의 별점이 전달받은 rating과 같은지 비교
    #
    # .sum()
    # → True를 1, False를 0으로 계산하여 True의 개수를 구함
    #
    # int()
    # → 결과를 일반 파이썬 정수로 변환
    return int((reviews['rating'] == rating).sum())


def get_weather_info(region, day):
    """
    전달받은 지역의 가상 날씨를 지정한 일수만큼 반환한다.

    실제 날씨 API를 사용하는 함수가 아니라,
    맑음, 흐림, 비 중 하나를 무작위로 선택하는 연습용 함수이다.
    """

    # 날짜별 날씨를 저장할 빈 리스트
    results = []

    # day에 전달된 숫자만큼 반복
    for _ in range(day):
        # 세 가지 날씨 중 하나를 무작위로 선택하여 리스트에 추가
        weather = random.choice(['맑음', '흐림', '비'])
        results.append(weather)

    # 함수 실행 결과를 문자열로 반환
    return f'{region} 지역의 {day}일 날씨 예보입니다. {results}'


# 함수가 정상적으로 실행되는지 직접 확인
print(count_reviews_by_rating(3))
print(get_weather_info('서울', 3))


# =========================================================
# 2단계: 모델에게 사용할 수 있는 도구 설명
# =========================================================

# 여기서는 실제 함수를 넣는 것이 아니라,
# 모델이 이해할 수 있도록 다음 내용을 설명함
#
# - 도구의 이름
# - 도구가 하는 일
# - 도구가 받을 인자
# - 각 인자의 자료형
#
# 모델은 이 설명을 읽고 어떤 함수를 호출할지 결정함
tools = [
    {
        # 사용자가 만든 함수를 도구로 제공한다는 뜻
        'type': 'function',

        # 첫 번째 도구의 상세 설명
        'function': {
            # 실제 파이썬 함수 이름과 같아야 함
            'name': 'count_reviews_by_rating',

            # 모델이 어떤 상황에서 이 도구를 선택할지 판단하는 설명
            'description': '특정 별점 1~5에 해당하는 리뷰 개수를 센다.',

            # 함수가 받을 인자의 구조를 JSON Schema로 설명
            'parameters': {
                # 여러 값을 키와 값으로 받으므로 object 사용
                'type': 'object',

                # 함수가 받을 수 있는 인자 목록
                'properties': {
                    'rating': {
                        # rating은 정수여야 함
                        'type': 'integer',

                        # 모델이 rating 값을 만드는 데 참고할 설명
                        'description': '조회할 별점. 1부터 5 사이의 정수'
                    }
                },

                # rating은 반드시 전달해야 하는 인자
                'required': ['rating']
            }
        }
    },
    {
        'type': 'function',

        # 두 번째 도구의 상세 설명
        'function': {
            # 실제 날씨 함수의 이름과 같아야 함
            'name': 'get_weather_info',

            # 모델이 날씨 질문에서 이 도구를 선택할 수 있도록 설명
            'description': (
                '특정 지역의 가상 날씨 예보를 '
                '오늘부터 지정한 일수만큼 알려준다. '
                '한 달은 30일로 처리한다.'
            ),

            # get_weather_info()가 받을 인자 설명
            'parameters': {
                'type': 'object',

                'properties': {
                    'region': {
                        # 지역명은 문자열
                        'type': 'string',
                        'description': '조회할 지역명. 예: 서울, 대전, 부산'
                    },
                    'day': {
                        # 조회 일수는 정수
                        'type': 'integer',
                        'description': '오늘부터 조회할 일수. 한 달은 30일'
                    }
                },

                # 두 인자가 모두 반드시 필요함
                'required': ['region', 'day']
            }
        }
    }
]


# tools에 두 도구가 올바르게 들어갔는지 확인
print('첫 번째 도구:', tools[0]['function']['name'])
print('두 번째 도구:', tools[1]['function']['name'])


# =========================================================
# 3단계: 사용자 질문을 모델에게 전달
# =========================================================

# 이 질문에는 두 가지 요청이 함께 들어 있음
#
# 1. 별점 3점 리뷰 개수
# 2. 대전의 한 달 날씨
#
# 따라서 모델은 두 도구를 모두 선택할 수 있음
messages = [
    {
        'role': 'user',
        'content': '리뷰 평점 3점의 데이터 개수와 대전의 한 달 날씨를 알려줘.'
    }
]


# tools를 함께 전달하여 모델이 필요한 도구를 선택하도록 함
resp = client.chat.completions.create(
    model='gpt-4o-mini',
    messages=messages,
    tools=tools,

    # 일반 답변 또는 도구 호출 중 모델이 자동으로 선택
    tool_choice='auto'
)


# 첫 번째 모델 응답 메시지 꺼내기
#
# 일반 질문이면:
# msg.content에 답변이 들어 있음
#
# 도구가 필요하면:
# msg.tool_calls에 도구 호출 정보가 들어 있음
msg = resp.choices[0].message


# =========================================================
# 4단계: 모델이 도구를 선택했는지 확인
# =========================================================

if not msg.tool_calls:
    # tool_calls가 없으면 모델이 도구가 필요하지 않다고 판단한 것
    print('도구 없이 바로 답:', msg.content)

else:
    # =====================================================
    # 5단계: 디스패처를 이용한 도구 라우팅
    # =====================================================

    # 모델이 반환하는 것은 함수 이름 문자열임
    #
    # 예:
    # 'count_reviews_by_rating'
    # 'get_weather_info'
    #
    # dispatch는 함수 이름과 실제 파이썬 함수를 연결함
    dispatch = {
        'count_reviews_by_rating': count_reviews_by_rating,
        'get_weather_info': get_weather_info
    }

    # 모델이 선택한 전체 도구 호출 내용 확인
    print('전체 도구 호출:', msg.tool_calls)

    # =====================================================
    # 6단계: 모델이 요청한 도구 호출 기록 만들기
    # =====================================================

    # 여러 개의 도구 호출 정보를 모을 빈 리스트
    assistant_tool_calls = []

    # 모델이 요청한 모든 도구를 하나씩 확인
    for call in msg.tool_calls:
        assistant_tool_calls.append({
            # 각각의 도구 호출을 구분하는 고유 ID
            'id': call.id,

            # 호출 유형은 function
            'type': 'function',

            # 모델이 선택한 함수 이름과 인자
            'function': {
                'name': call.function.name,
                'arguments': call.function.arguments
            }
        })

    # 모델이 어떤 도구들을 호출했는지 대화 기록에 추가
    #
    # 중요한 점:
    # 도구가 두 개여도 assistant 메시지는 한 번만 추가하고,
    # 그 안의 tool_calls 리스트에 모든 호출을 함께 넣음
    messages.append({
        'role': 'assistant',
        'content': msg.content,
        'tool_calls': assistant_tool_calls
    })

    # =====================================================
    # 7단계: 모델이 선택한 모든 도구를 실제로 실행
    # =====================================================

    for call in msg.tool_calls:
        # 모델이 선택한 함수 이름
        function_name = call.function.name

        # 모델이 만든 함수 인자는 JSON 문자열 형태임
        arguments_json = call.function.arguments

        print('모델이 선택한 함수:', function_name)
        print('전달한 인자 JSON:', arguments_json)

        # JSON 문자열을 파이썬 딕셔너리로 변환
        #
        # 예:
        # '{"rating": 3}'
        #          ↓
        # {'rating': 3}
        args = json.loads(arguments_json)

        print('변환된 인자 딕셔너리:', args)

        # dispatch에서 함수 이름에 해당하는 실제 함수를 찾음
        selected_function = dispatch[function_name]

        # **args로 딕셔너리의 값을 함수 인자로 풀어서 전달
        #
        # args가 {'rating': 3}이라면:
        # selected_function(**args)
        #
        # 실제 실행:
        # count_reviews_by_rating(rating=3)
        tool_result = selected_function(**args)

        print('도구 실행 결과:', tool_result)

        # =================================================
        # 8단계: 각 도구의 실행 결과를 대화 기록에 추가
        # =================================================

        # tool_call_id를 사용해
        # 이 결과가 어떤 도구 호출에 대한 결과인지 연결함
        messages.append({
            'role': 'tool',
            'tool_call_id': call.id,

            # 도구 결과는 문자열로 변환하여 전달
            'content': str(tool_result)
        })

    # =====================================================
    # 9단계: 모든 도구 결과로 최종 답변 생성
    # =====================================================

    # messages에는 현재 다음 내용이 들어 있음
    #
    # 1. 사용자의 원래 질문
    # 2. 모델이 요청한 여러 도구 호출
    # 3. 첫 번째 도구의 실행 결과
    # 4. 두 번째 도구의 실행 결과
    #
    # 모델은 이 기록을 바탕으로 사용자에게 보여줄 문장을 작성함
    second = client.chat.completions.create(
        model='gpt-4o-mini',
        messages=messages
    )

    # 최종 자연어 답변 추출
    final_answer = second.choices[0].message.content

    print('최종 답변:', final_answer)
```
